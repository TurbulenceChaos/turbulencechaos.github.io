\begin{definition}[types]~

\end{verbatim}

              \begin{verbatim}
print("\t")
print(r"\t") # raw

# multiple lines
"""\
line 1
line 2"""

# concat and repeat
3 * "a" + "b" # 'aaab'

# index and slice
str="Python"
#  | P | y | t | h | o | n |
#  0   1   2   3   4   5   6
# -6  -5  -4  -3  -2  -1
str[2]   # 't', str[2] = "a", error, immutable
str[1:2] # 'y'
len(str) # 6
\end{verbatim}
        \item list
    \end{itemize}
