*** Scope of a name
#+name: cpp-scope
#+begin_src cpp
int reused = 10;
int main ()
{
    std::cout << reused << "\n";
    int reused = 0;
    std::cout << reused << "\n";
    std::cout << ::reused << "\n";

    int i = 100, sum = 0;
    for(int i = 0; i != 10; ++i)
        sum += i;
    std::cout << i << " " << sum << "\n";

    return 0;

* Linked List
#+name: cpp-linked-list
#+begin_src cpp
struct Node
{
    int val;
    Node *next;
    Node(int x) : val(x), next(nullptr) {}
};

int main()
{
    Node *n1 = new Node(1);
    Node *n2 = new Node(2);
    n1->next = n2;
    while (n1 != nullptr)
    {
        std::cout << n1->val << std::endl;
        n1 = n1->next;
    }
    return 0;
}
#+end_src

#+RESULTS: cpp-linked-list
| 1 |
| 2 |
