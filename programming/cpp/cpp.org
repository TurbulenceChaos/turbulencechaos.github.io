* Linked List
#+name: cpp-linked-list
#+begin_src cpp
struct Node
{
    int val;
    Node *next;
    Node(int x) : val(x), next(nullptr) {}
};

int main()
{
    Node *n1 = new Node(1);
    Node *n2 = new Node(2);
    n1->next = n2;
    while (n1 != nullptr)
    {
        std::cout << n1->val << std::endl;
        n1 = n1->next;
    }
    return 0;
}
#+end_src

| 2 |
