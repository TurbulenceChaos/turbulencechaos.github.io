\section{Introduction}
Lean is a \textbf{expression-oriented} functional language with dependent types \cite{CHRISTIANSEN_2023_FUNCTIONAL_PROGRAMMING_IN_LEAN}.

\subsection{Ealuating Expressions}
\begin{itemize}
    \item every expression must have a type before it can be evaluated
    \item once given a value, variables cannot be reassigned
    \item parentheses are necessary when a function's argument is to be computed via another function call, i.e. $5*(1+2)$
\end{itemize}

\subsection{Types}
\begin{definition}[type]~
    \begin{itemize}
        \item \verb|Nat|: default type for non-negative integer literals
        \item \verb|Int|
        \item \verb|Float|
        \item \verb|String|
        \item \verb|Bool|
    \end{itemize}
\end{definition}

\begin{example}~
    \begin{verbatim}
#eval (1 - 2 : Nat) -- 0
#check (1 - 2 : Int) -- check the type of an expression
\end{verbatim}
\end{example}

\subsection{Functions and Definitions}
\begin{definition}[definition]~
    \begin{verbatim}
def var : type := val
\end{verbatim}
\end{definition}

\begin{example}~
    \begin{verbatim}
def lean : String := "Lean"
#eval String.append "hello " lean
\end{verbatim}
\end{example}

\subsubsection{Defining Functions}
\begin{definition}[function]~
    \begin{verbatim}
def func (arg1 : type) ... (argn : type) : type := body
\end{verbatim}
\end{definition}

\begin{example}~
    \begin{verbatim}
def maximum (n : Nat) (m : Nat) : Nat := if n < m then m else n
#eval maximum 1 5
\end{verbatim}
\end{example}

\begin{definition}[currying]
    \verb|Nat -> (Nat -> Nat)|
\end{definition}

\subsection{Structures}
\begin{definition}[structure]~
    \begin{verbatim}
structure Point where
x : Float
y : Float
def origin : Point := {x:=0.0,y:=0.0}
#eval origin.x
#check {x:=0.0,y:=0.0:Point}
\end{verbatim}
\end{definition}

\begin{definition}[updating structures]
    \begin{verbatim}
def zeroX (p : Point) := {p with x:=0}   
\end{verbatim}
\end{definition}

\begin{definition}[constructor]~
    \begin{verbatim}
structure Point where
point::
x : Float
y : Float
#check Point.point 1.0 2.0
\end{verbatim}
\end{definition}

\begin{definition}[accessor]~
