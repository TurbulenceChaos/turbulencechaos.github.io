Lean is an expression tactic functional language based on \textbf{dependent type theory}.

\subsection{Overview}
Every expression has a type (printed by \verb|#check|).

\begin{definition}[mathematical object]~
    \begin{verbatim}
#check 2 + 2 -- Nat

def f (x : Nat) :=
  x+3

#check f -- Nat -> Nat
\end{verbatim}
\end{definition}

\begin{definition}[mathematical statement: Prop]~
    \begin{verbatim}
#check 2 + 2 = 4 -- Prop

def FermatLastTheorem :=
  \all x y z n : Nat, n > 2 \and x * y *z \neq 0 \to x^n + y^n \neq z^n

#check FermatLastTheorem -- Prop
\end{verbatim}
\end{definition}

\begin{definition}[proof of the proposition]~
    \begin{verbatim}
theorem easy : 2 + 2 = 4 :=
  rfl

#check easy -- 2 + 2 = 4

theorem hard : FermatLastTheorem :=
  sorry

#check hard -- FermatLastTheorem
\end{verbatim}
\end{definition}

Construct an expression of type \verb|FermatLastTheorem| and make Lean accepts it as a term of that type.

\subsubsection{Pattern Matching}
\begin{definition}[pattern matching]~
    \begin{verbatim}
def isZero (n:Nat) : Bool :=
match n with
| Nat.zero => true
| Nat.succ k => false
#eval isZero Nat.zero

def pred (n : Nat) : Nat :=
match n with
| Nat.zero => Nat.zero
| Nat.succ k => k
#eval pred 5

def depth (p:Point3D) :Float :=

| {x:=h,y:=w,z:=d} => d
#eval depth {x:=1,y:=2,z:=3}
\end{verbatim}

\subsubsection{Recursive Functions}
\begin{definition}[recursive definition]
    Definitions that refer to the name being defined are called recursive definition.
\end{definition}

\begin{definition}[structural recursion]~
    \begin{verbatim}
def even (n:Nat) : Bool :=
match n with
| Nat.zero => true
| Nat.succ k => not (even k)
#eval even 2

-- addition = interated Nat.succ
def plus (n:Nat) (k:Nat) : Nat :=
match k with
| Nat.zero => n
| Nat.succ k' => Nat.succ (plus n k')
#eval plus 1 2

-- multiplication = interated addition
def times (n:Nat) (k:Nat) : Nat :=
match k with
| Nat.zero => 0
| Nat.succ k' => plus n (times n k')
#eval times 1 2

-- subtraction = interated predecessor
def minus (n:Nat) (k:Nat) : Nat :=
match k with
| Nat.zero => n
| Nat.succ k' => Nat.pred (minus n k')
#eval minus 2 1
\end{verbatim}
\end{definition}

\begin{remark}
    Lean ensures by default that every recursive function will eventually reach a base case to avoid accidental infinite loops
\end{remark}

\begin{example}~
    \begin{verbatim}
def div (n:Nat) (k:Nat) : Nat :=
if n < k then
0
else Nat.succ (div (n-k) k)
\end{verbatim}
\end{example}

\subsection{Polymorphism}
\begin{definition}[polymorphism]
    polymorphism refers to datatypes and definitions that take types as arguments, where \textbf{types} are ordinary \textbf{expressions} in Lean.
\end{definition}

\begin{example}~
    \begin{verbatim}
-- structure take types as arguments
structure PPoint (\alpha : Type) where
x : \alpha
y : \alpha
def natOrigin : PPoint Nat := {x:=Nat.zero,y:=Nat.zero}
#check natOrigin

-- definition take types as arguments
def replaceX (\alpha : Type) (point : PPoint \alpha) (newX : \alpha) : PPoint \alpha :=
{point with x:=newX}
#check replaceX
#eval replaceX Nat natOrigin 2

-- argument is a sign
inductive Sign where
| pos
| neg
def posOrNegThree (s : Sign) :
match s with | Sign.pos => Nat | Sign.neg => Int :=
match s with
| Sign.pos => (3:Nat)
| Sign.neg => (-3:Int)
#check posOrNegThree Sign.pos
\end{verbatim}
\end{example}

\begin{remark}
    It is customary to use Greek letters to name type arguments.
\end{remark}

\subsubsection{Linked List Datatype}
\begin{definition}[List]
    A \verb|List a| can be built with either \verb|nil| or \verb|cons|.
    A list that contains $n$ entries contains $n$ \verb|cons| constructors, the last of which has \verb|nil| as its tail.
    \begin{verbatim}
inductive List (\alpha : Type) where
| nil : List \alpha
| cons : \alpha -> List \alpha -> List \alpha
def primesUnder10 : List Nat := [2,3,5,7]
#eval List.cons 2 (List.cons 3 (List.cons 5 (List.cons 7 List.nil)))

-- functions that consume Lists
def length (\alpha : Type) (xs : List \alpha) : Nat :=
match xs with
| List.nil => Nat.zero -- [] for List.nil
| List.cons y ys => Nat.succ (length \alpha ys) -- y :: ys for List.cons
#eval length String ["hello", "lean"]
\end{verbatim}
\end{definition}

\subsubsection{Implicit Arguments}
\begin{definition}[implicit type]~
    \begin{verbatim}
def length {\alpha : Type} (xs : List \alpha) : Nat :=
match xs with
| [] => Nat.zero
| y :: ys => Nat.succ (length ys)
#eval length ["hello", "lean"]
-- List.length
#eval ["hello", "lean"].length
\end{verbatim}
\end{definition}

\subsubsection{More Built-In Structures and Inductive Datatypes}
\begin{definition}[option]
    List is empty or has a first entry.
    \begin{verbatim}
inductive Option (\alpha : Type) : Type where
| none : Option \alpha
| some (val : \alpha) : Option \alpha

def List.head? {\alpha : Type} (xs : List \alpha) : Option \alpha :=
match xs with
| [] => none
| y :: _ => some y
#eval [1,2,3].head?
#eval ([] : List Int).head?
\end{verbatim}
\end{definition}

\begin{remark}
    Type was unavailable because the empty list does not have any entries from which the type can be found.
\end{remark}

name convention of suffixes for operations:
\begin{itemize}
    \item ? for operations that might fail
    \item ! for crash due to invalid input
    \item D for default value when the operation would otherwise fail
\end{itemize}

\begin{definition}[Prod]
    Constructs a pair.
    \begin{verbatim}
structure Prod (\alpha : Type) (\beta : Type) : Type where
fst : \alpha
snd : \beta
-- def fives : Prod String Int := {fst:="five",snd:=5}
-- def fives : String \times Int := {fst:="five",snd:=5}
def fives : String \times Int := ("five",5)
\end{verbatim}
\end{definition}

\begin{definition}[Sum]
    Allow a choice between values of two different types.
    \begin{verbatim}
inductive Sum (\alpha : Type) (\beta : Type) : Type where
| inl : \alpha -> Sum \alpha \beta
| inr : \beta -> Sum \alpha \beta

-- def PetName : Type := Sum String String
def PetName : Type := String \oplus String

def animals : List PetName :=
[Sum.inl "Spot", Sum.inr "Tiger",Sum.inl "Fifi",Sum.inl "Rex",Sum.inr "Floor"]

def howManyDogs (pets : List PetName) : Nat :=
match pets with
| [] => 0
| Sum.inl _ :: morePets => howManyDogs morePets + 1
| Sum.inr _ :: morePets => howManyDogs morePets

#eval howManyDogs animals
\end{verbatim}
\end{definition}
