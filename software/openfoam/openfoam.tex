\section{Introduction}

\begin{definition}[openfoam]
    \url{https://openfoam.org/resources/}

    OpenFOAM is a free, open source computational fluid dynamics software
    to solve complex fluid flows involving turbulence and solid dynamics.
\end{definition}

Components of OpenFOAM:
\begin{itemize}
    \item solvers: perform calculation in fluid
    \item utilities: config simulation, geometry, mesh, and process results
\end{itemize}

\section{Mesh Generation and Conversion}
\begin{definition}[mesh]
    OpenFOAM always operates in a three dimensional Cartesian coordinate system
    and all geometries are generated in three dimensions.

    OpenFOAM solves cases in two dimensions by specifying \verb|empty| boundary condition
    on boundaries normal to the third dimension for which no solution is required.
\end{definition}

\begin{definition}[blockMesh]
    blockMesh generates meshes from a \verb|blockMeshDict| file
    located in \verb|system| directory for a given case.
    \small
    \begin{verbatim}
// -*- C++ -*-
// Website: https://openfoam.org
// Version: openfoam-dev

FoamFile // sub-dictionary
{
    format   ascii
    class    dictionary
    location "system"
    object   blockMeshDict
}

units [mm]; // optional
// the generated mesh is written in the base units for the underlying unit system, i.e. meter

vertices
(
    (x1 y1 z1) // vertex 1
    ...
);

blocks
(
    hex (v1 ... v8)
    (cells_x cells_y cells_z)
    simpleGrading (g_x g_y g_z)
    ...
);

boundary
(
    name
    {
        type T;
        faces
        (
            (v1 ... v4)
            ...
        );
    }
);
\end{verbatim}
\end{definition}
