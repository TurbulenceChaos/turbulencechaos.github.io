\section{Curved Path Construction}
\begin{definition}[control points]~
    \begin{verbatim}
(start_point) .. controls (control_point_1) and (control_point_2) .. (end_point)
\end{verbatim}
\end{definition}

\begin{LTXexample}[pos=b]
    \begin{tikzpicture}[scale=2]
        \filldraw [gray]
        (0,0) circle (2pt)
        (1,1) circle (2pt)
        (2,1) circle (2pt)
        (2,0) circle (2pt);
        \draw [thick] (0,0) .. controls (1,1) and (2,1) .. (2,0);
        \draw [-latex] (0,0) -- node[sloped,above] {tangent} (1,1);
        \draw [-latex] (2,0) -- node[sloped,below] {tangent} (2,1);
    \end{tikzpicture}
\end{LTXexample}
