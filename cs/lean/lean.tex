\section{Calculating}
\begin{definition}
\begin{verbatim}
mul_assoc a b c => a * b * c = a * (b * c)
\end{verbatim}
\end{definition}

\begin{definition}
\begin{verbatim}
mul_comm a b => a * b = b * a
\end{verbatim}
\end{definition}

\begin{proposition}
  Let $a, b, c$ be real numbers, then  
  \begin{equation*}
    a b c = b (a c)
  \end{equation*}
\end{proposition}

\begin{proof}~
\begin{verbatim}
example (a b c : ℝ) : a * b * c = b * (a * c) := by
  rw [mul_comm a] -- b * a * c   = b * (a * c)
  rw [mul_assoc]  -- b * (a * c) = b * (a * c)
\end{verbatim}
\end{proof}

\begin{proposition}
  Let $a, b, c$ be real numbers.
  If $a b = c d$ and $e = f$, then
  \begin{equation*}
    a (b e) = c (d f)
  \end{equation*}
\end{proposition}

\begin{proof}~
\begin{verbatim}
variable (a b c d e f : ℝ)

example (h : a * b = c * d)
(h' : e = f) :
a * (b * e) = c * (d * f) := by
 rw [h']          -- a * (b * f) = c * (d * f)
 rw [← mul_assoc] -- a * b * f   = c * (d * f)
 rw [h]           -- c * d * f   = c * (d * f)
 rw [mul_assoc]   -- c * (d * f) = c * (d * f)
\end{verbatim}
\end{proof}

\begin{definition}
\begin{verbatim}
two_mul a => 2 * a = a + a
\end{verbatim}
\end{definition}

\begin{definition}
\begin{verbatim}
add_mul a b c => (a + b) * c = a * c + b * c
\end{verbatim}
\end{definition}

\begin{definition}
\begin{verbatim}
mul_add a b c => a * (b + c) = a * b + a * c
\end{verbatim}
\end{definition}

\begin{definition}
\begin{verbatim}
add_assoc a b c => a + b + c = a + (b + c)
\end{verbatim}
\end{definition}

\begin{proposition}
  Let $a, b$ be real numbers.
  \begin{equation*}
    (a + b) (a + b) = a a + 2 (a b) + b b
  \end{equation*}
\end{proposition}

\begin{proof}~
\begin{verbatim}
example (a b : ℝ) : (a + b) * (a + b) = a * a + 2 * (a * b) + b * b := by
  rw [mul_add]           -- (a + b) * a + (a + b) * b       = a * a + 2 * (a * b) + b * b
  rw [add_mul]           -- a * a + b * a + (a + b) * b     = a * a + 2 * (a * b) + b * b
  rw [add_mul]           -- a * a + b * a + (a * b + b * b) = a * a + 2 * (a * b) + b * b
  rw [mul_comm b a]      -- a * a + a * b + (a * b + b * b) = a * a + 2 * (a * b) + b * b
  rw [← add_assoc]       -- a * a + a * b + a * b + b * b   = a * a + 2 * (a * b) + b * b
  rw [add_assoc (a * a)] -- a * a + (a * b + a * b) + b * b = a * a + 2 * (a * b) + b * b
  rw [← two_mul]         -- a * a + 2 * (a * b) + b * b     = a * a + 2 * (a * b) + b * b
\end{verbatim}
\end{proof}
