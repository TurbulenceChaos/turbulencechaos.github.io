\section{Calculating}
\begin{definition}
  \label{DEF:MUL_ASSOC}
\begin{verbatim}
[mul_assoc]
mul_assoc a b c => a * b * c = a * (b * c)
\end{verbatim}
\end{definition}

\begin{definition}
  \label{DEF:MUL_COMM}
\begin{verbatim}
[mul_comm]
mul_comm a b => a * b = b * a
\end{verbatim}
\end{definition}
