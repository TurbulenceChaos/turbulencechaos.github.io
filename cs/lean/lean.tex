#check (mul_assoc : ∀ a b c : ℝ, a * b * c = a * (b * c))

\begin{example}
  Let $a, b, c$ be real numbers.
  If $a b = c d$ and $e = f$, then
  \begin{equation*}
    a (b e) = c (d f)
  \end{equation*}
\end{example}

\begin{proof}~
  \begin{verbatim}
variable (a b c d e f : ℝ)

example (h : a * b = c * d) (h' : e = f) : a * (b * e) = d * f * c :=
  calc
    a * (b * e) = a * (b * f) := by rw [h']
      _         = a * b * f   := by rw [← mul_assoc]
      _         = c * d * f   := by rw [h]
      _         = c * (d * f) := by rw [mul_assoc]
      _         = d * f * c   := by rw [mul_comm]
\end{verbatim}
\end{proof}

#check (add_assoc : ∀ a b c : ℝ, a + b + c = a + (b + c))

\end{definition}

\begin{proposition}
  Let $a, b$ be real numbers, then
  \begin{equation*}
    (a + b) (a + b) = a a + 2 (a b) + b b
  \end{equation*}
\end{proposition}

\begin{proof}~
  \begin{verbatim}
example (a b : ℝ) : (a + b) * (a + b) = a * a + 2 * (a * b) + b * b :=
  calc
    (a + b) * (a + b) = (a + b) * a + (a + b) * b       := by rw [mul_add]
    _                 = a * a + b * a + (a + b) * b     := by rw [add_mul]
    _                 = a * a + b * a + (a * b + b * b) := by rw [add_mul]
    _                 = a * a + a * b + (a * b + b * b) := by rw [mul_comm b a]
    _                 = a * a + a * b + a * b + b * b   := by rw [← add_assoc]
    _                 = a * a + (a * b + a * b) + b * b := by rw [add_assoc (a * a)]
    _                 = a * a + 2 * (a * b) + b * b     := by rw [← two_mul]
\end{verbatim}
\end{proof}

\begin{proposition}
  Let $a, b$ be real numbers, then
  \begin{equation*}
    (a + b) (a - b) = a^{2} - b^{2}
  \end{equation*}
\end{proposition}

\begin{proof}~
  \begin{verbatim}
example (a b : ℝ) : (a + b) * (a - b) = a ^ 2 - b ^ 2 :=
  calc
    (a + b) * (a - b) = a * (a - b) + b * (a - b)       := by rw [add_mul]
    _                 = a * a - a * b + b * (a - b)     := by rw [mul_sub]
    _                 = a * a - a * b + (b * a - b * b) := by rw [mul_sub]
    _                 = a * a - a * b + b * a - b * b   := by rw [add_sub]
    _                 = a * a - (a * b - b * a) - b * b := by rw [sub_add]
    _                 = a * a - (a * b - a * b) - b * b := by rw [mul_comm b a]
    _                 = a * a - 0 - b * b               := by rw [sub_self]
    _                 = a * a - b * b                   := by rw [sub_zero]
    _                 = a ^ 2 - b * b                   := by rw [← pow_two]
    _                 = a ^ 2 - b ^ 2                   := by rw [← pow_two]
\end{verbatim}
\end{proof}

\begin{proposition}
  Let $a, b, c, d$ be real numbers.
  If $c = d a + b$ and $b = a d$, then
  \begin{equation*}
    c = 2 a d
  \end{equation*}
\end{proposition}

\begin{proof}~
  \begin{verbatim}
variable (a b c d : ℝ)

example (h : c = d * a + b) (h' : b = a * d) : c = 2 * a * d := by
  rw [h'] at h
  rw [mul_comm d a] at h
  rw [← two_mul] at h
  rw [← mul_assoc] at h
  exact h

example (h : c = d * a + b) (h' : b = a * d) : c = 2 * a * d :=
  calc
    c = d * a + b     := h
    _ = d * a + a * d := by rw [h']
    _ = a * d + a * d := by rw [mul_comm]
    _ = 2 * (a * d)   := by rw [← two_mul]
    _ = 2 * a * d     := by rw [mul_assoc]
\end{verbatim}
\end{proof}

\end{definition}

\begin{proposition}
  Let $a, b, c$ be natural numbers.
  If $a + b = c$, then
  \begin{equation*}
    (a + b) (a + b) = a c + b c
  \end{equation*}
\end{proposition}

\begin{proof}~
  \begin{verbatim}
example (a b c : ℕ) (h : a + b = c) : (a + b) * (a + b) = a * c + b * c := by
  calc
    (a + b) * (a + b) = (a + b) * c   := by nth_rw 2 [h]
    _                 = a * c + b * c := by rw [add_mul]
\end{verbatim}
\end{proof}

\begin{example}~
  \begin{verbatim}
namespace MyRing

variable {R : Type*} [Ring R]

theorem neg_add_cancel_left (a b : R) : -a + (a + b) = b :=
  calc
    -a + (a + b) = -a + a + b := by rw [← add_assoc]
    _ = 0 + b                 := by rw [neg_add_cancel]
    _ = b                     := by rw [zero_add]

theorem neg_add_cancel_right (a b : R) : a + b + -b = a :=
  calc
    a + b + -b = a + (b + -b) := by rw [add_assoc]
      _        = a + 0        := by rw [add_neg_cancel]
      _        = a            := by rw [add_zero]

theorem add_left_cancel {a b c : R} (h : a + b = a + c) : b = c :=
  calc
    b = -a + (a + b) := by nth_rw 1 [← neg_add_cancel_left a b]
    _ = -a + (a + c) := by rw [h]
    _ = c            := by rw [neg_add_cancel_left]

theorem add_right_cancel {a b c : R} (h : a + b = c + b) : a = c :=
  calc
    a = a + b + -b := by nth_rw 1 [← neg_add_cancel_right a b]
    _ = c + b + -b := by rw [h]
    _ = c          := by rw [neg_add_cancel_right]

end MyRing
\end{verbatim}
\end{example}

\begin{proposition}
  Let $a$ be a real number, then
  \begin{gather*}
    a \times 0 = 0 \\
    0 \times a = 0
  \end{gather*}
\end{proposition}

\begin{proof}~
  \begin{verbatim}
theorem mul_zero (a : R) : a * 0 = 0 := by
  have h : a * 0 + a * 0 = a * 0 + 0 := by
    rw [← mul_add]
    rw [add_zero]
    rw [add_zero]
  rw [add_left_cancel h]

theorem zero_mul (a : R) : 0 * a = 0 := by
  have h : 0 * a + 0 * a = 0 * a + 0 :=
    calc
      0 * a + 0 * a = (0 + 0) * a := by rw [← add_mul]
        _           = 0 * a       := by rw [add_zero]
        _           = 0 * a + 0   := by nth_rw 1 [← add_zero (0 * a)]
  rw [add_left_cancel h]
\end{verbatim}
\end{proof}
