\begin{definition}[simple type theory]
    Build new types out of others.
    \begin{verbatim}
type → type -- denotes the type of function from type to type
type → type → type -- equivalent to type → (type → type) that
-- takes type and return another function that takes type and return type

type × type -- denotes the type of pairs consisting of type paired with type
(m, n) denotes the ordered pair of m and n which is of type × type
(m, n).1 : type
(m, n).2 : type
\end{verbatim}
\end{definition}

\section{Tactics}
\begin{tactic}[reflexivity of equality]
    \verb|ref| proves goals of the form \verb|X = X|.
\end{tactic}

\begin{tactic}[rewrite]
    If \verb|h| is a proof of an equality \verb|X = Y|,
    then \verb|rw [h]| replace all \verb|X| with \verb|Y|.
    \begin{verbatim}
repeat rw [h] -- rewrite at once
rw [← h] -- turn Y into X
\end{verbatim}
\end{tactic}

\begin{definition}[$\mathbb{N}$]
    $\mathbb{N}$ is the natural numbers defined by \verb|0 : ℕ| and \verb|succ (0 : ℕ) : ℕ|.
\end{definition}

\begin{theorem}[successor]
    The successor of \verb|n| means the number after \verb|n|.
    \begin{verbatim}
two_eq_succ_one -- proof of 2 = succ 1
add_succ x d : x + succ d = succ (x + d)
\end{verbatim}
\end{theorem}

\begin{theorem}[adding 0]
    Define \verb|a + 0| to be \verb|a| for any number \verb|a|.
    \begin{verbatim}
add_zero   -- proof of ? + 0 = ?
add_zero x -- proof of x + 0 = x
\end{verbatim}
\end{theorem}
