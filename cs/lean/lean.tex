\section{Introduction}
\begin{definition}[lean]
    Lean is an expression-oriented functional language based on
    \href{https://lean-lang.org/definition_proving_in_lean4/}{dependent type theory}.
\end{definition}

\section{Dependent Type Theory}
\begin{definition}[definition]
    Declare a new named constant object.
    \begin{verbatim}
def c : type := expr
-- auxiliary commands begin with # symbol
#check c   -- check type
#eval expr -- evaluate expression
#print c   -- view definition

-- a name defined using def is semireducible
def T := type
def c : T := ... -- error

-- local definition
-- strong means of abbreviation
#check let c := ...; expr -- replace c in expr
\end{verbatim}
\end{definition}

\begin{definition}[type]
    Every expression has an associated type.
    \begin{verbatim}
Nat
#check Nat.succ
#check Nat.add
Int
Float
String
Bool -- true or false

List -- List type
#check List.nil
#check []
#check [1, 2, 3]

#check List.cons
#check List.cons 1 []
#check 1 :: 2 :: 3 :: []

#check List.map
#check List.map func [1, 2, 3]
#check [1, 2, 3].map func

Type -- abbreviation for Type 0
-- infinite hierarchy of types
-- larger universe of types contains lower universe of types as elements
-- Type : Type 1
-- Type n : Type n + 1
universe u -- polymorphic constant
Type u
\end{verbatim}
\end{definition}

\begin{definition}[simple type theory]
    Build new types out of others.
    \begin{verbatim}
type → type        -- denotes the type of function
type → type → type -- equivalent to type → (type → type)
-- arrows associate to the right in type expressions

type × type    -- denotes the type of Cartesian product
Prod type type -- alternative notation
(m, n)         -- denotes the ordered pair of m and n which is of type type × type
(m, n).1       -- first component
(m, n).2       -- second component
\end{verbatim}
\end{definition}

\begin{definition}[dependent type theory]
    Dependent type theory extends simple type theory that
    types themselves are objects which also have a type.
\end{definition}

\begin{example}[polymorphic]
    A function is polymorphic that takes the type to be the (first) argument.
    \begin{verbatim}
def cons (α : Type) (a : α) (as : List α) : List α :=
  List.cons a as

#check cons Nat
#check cons

-- dependent function type or dependent arrow type
-- the type of the value returned by function depends on its input
-- dependent Cartesian product
def f (α : Type u) (β : α → Type v) (a : α) (b : β a) : (a : α) × β a := ⟨a, b⟩
def g (α : Type u) (β : α → Type v) (a : α) (b : β a) : Σ a : α, β a := Sigma.mk a b
\end{verbatim}
\end{example}

\section{Function Abstraction}
\begin{definition}[lambda function]
    Creating a function from another expression is a process called lambda abstraction.
    \begin{verbatim}
#check fun (param_1 : type) ... => expr
-- lambda function is an object of type type → ...
-- λ and fun are the same

#eval (λ (param_1 : type) ... ) arg_1

-- declare a new named object
def f : type → ... := λ (param_1 : type) ... => expr
-- move lambda-abstracted variables to the left of the colon
-- with input parameters and returning type
def f (param_1 : type) ... : type := expr

#eval f arg_1 ... arg_n -- call function
\end{verbatim}
\end{definition}

\begin{example}
    \begin{verbatim}
def compose (α β γ : Type) (g : β → γ) (f : α → β) (x : α) : γ := g (f x)
\end{verbatim}
\end{example}

\section{Variable and Sections}
\begin{definition}[variable]
    Instruct Lean to insert the declared variables as
    bounded variables in definitions that refer to them by name.
    \begin{verbatim}
variable (T : type) ...

variable (α β γ : Type)
variable (g : β → γ) (f : α → β)
variable (x : α)
def compose : γ := g (f x)
\end{verbatim}
\end{definition}

\begin{definition}[section]
    Limit the scope of a variable.
    \begin{verbatim}
section MySection
    ...
end MySection
\end{verbatim}
\end{definition}

\begin{definition}[namespace]
    Group definitions.
    \begin{verbatim}
namespace MyNamespace
    def c : type := ...
end MyNamespace

#check MyNamespace.c
open MyNamespace -- use shorter names
#check c
\end{verbatim}
\end{definition}
