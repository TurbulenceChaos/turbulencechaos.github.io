\subsection{Qualifier}
\begin{definition}[const]
    The compiler will replace the const var with its value.
    \begin{verbatim}
// const must be initialized
const int i = 1;

// const var is local to file by default
// must specify extern to be used in other files
// source.cpp
extern const i = 10;
// header.h
extern const i;
// main.cpp
#include "header.h"
i ...

// ref to const
// cannot change the object to which it bounds
const int i = 1;
const int &r = i;
const int &r2 = 2; // compiler: const int temp = 2; const int &r2 = temp;

// pointer to const
// cannot change the object to which it points
const int i = 1;
const int *p = &i;

// const pointer
// the address it holds cannot be changed
int i = 1;
int *const p = &i; // p always points to i
\end{verbatim}
\end{definition}

\begin{definition}[constant expr]
    A constant expression is an expr whose value cannot change at compile time.
    \begin{verbatim}
constexpr int i = size();   // ok only if size is a constexpr function
constexpr int *p = nullptr; // const pointer, constexpr applies to p
\end{verbatim}
\end{definition}

\begin{definition}[type alias]
    A synonym for a type.
    \begin{verbatim}
typedef int base;
using base = int;
typedef base *p;  // int*: pointer to int
p p1 = &i;

// const p: const modify the given type: int *const: constant pointer to int
const p p2 = 0; // int *const p2 = 0
const p *p3;    // int *const *p3
\end{verbatim}
\end{definition}

\begin{definition}[const level]~
    \begin{verbatim}
// top-level const: an object itself is const, i.e. const pointer
const int i = 1;
int *const p1 = &i;

// low-level const: point or refer to const object
const int *p2 = &i;
const int &r = i;

// when copying, top-level const is ignored,
// low-level const is never ignored
const int *const p3 = &i;
p2 = p3;
\end{verbatim}
\end{definition}

\section{Exprs}
\begin{definition}[expression]
    Composed of operators and operands.
    \begin{verbatim}
// unary operator: act one one operand
// & address-of
// * dereference

// binary operator: act on two operands1
// == equality
// * multiplication

// overloaded operator
// define operator in class

// lvalue and rvalue
// lvalue: can be assignment
// rvalue: cannot be assignment
// lvalue = ...                   return left-hand operand as lvalue
// address-of &lvalue             return a pointer to lvalue as rvalue
// deref * and subscript operator return lvalue
// ++iterator or --iterator       return lvalue
// decltype(lvalue)               return ref
// decltype(rvalue)               return the exact type of rvalue
decltype(*p) // int&
decltype(&i) // int*
decltype(&p) // int**

// compound expr with two or more operators
// precedence:
// operands of operators with higher precedence group tightly than
// operands of operators at lower precedence
// associativity: group operands with same precedence
// left associative: operators with same precedence group from left-most to right
// right associative: operators with same precedence group from right-most to left
// () force grouping

// arithmetic operator
// left associative
// return rvalue
// reminder: int % int
// return same sign as m
(-m) % n // -(m % n)
m % (-n) // -(m % n)

// logical and relational operator
// return bool val
// 0 return false
// non-zero return true
// guarantee eval order
expr1 && expr2 // eval expr2 if and only if expr1 is true
expr1 || expr2 // eval expr2 if and only if expr1 is false
// precedence:
// right associative
// 1. !
// left associative
// 2. <=, >=
// 3. ==, !=
// 4. &&
// 5. ||
i != j < k // i != (j < k)

// assignment operator
// low precedence
// right associative
// left-hand operand: a modifiable lvalue
// return left-hand operand lvalue
// compound assignment operator
// *=, /=, %=
a *= b;    // compound assignment: a eval only once
a = a * b; // ordinary assignment: a eval twice

// increment and decrement operator
// require lvalue operand
// prefix:  increment operand and return the changed object as lvalue
// postfix: increment operand and return object's original value as rvalue
// use postfix only when necessary
// precedence: ++-- > dereference
*b++ // *(b++)

// evaluation order

// member access operator
p->mem => (*p).mem
// precedence: . > dereference
pointer-> return lvalue
.mem return lvalue or rvalue same as mem

// conditional operator
cond ? expr1 : expr2;
// return lvalue if both expr1 and expr2 are lvalue
// otherwise return rvalue
cond ? expr1 : cond ? expr2 : expr3
// right associative
// low precedence, need ()

// bitwise operator
// hardware

// shift operator (IO operator)
// left associative
// precedence:
1. arithmetic operator
2. IO operator
3. relational, assignment, conditional operator

// sizeof operator
// right associative
// return constexpr size_t in byte
// 1 byte = 8 bits
sizeof(type)
sizeof(expr)
// array
sizeof(a) // sizeof(*a) * size

// comma operator
// guarantee eval order from left to right
// lowest precedence
// expr1, expr2
// return expr2 lvalue or rvalue
for (int v1=..., v2=...; cond; expr1, expr2)
    ...
\end{verbatim}
\end{definition}

\begin{definition}[IO objects]~
    \begin{verbatim}
#include <iostream>

\end{verbatim}
\end{definition}

\section{Statement}
\begin{definition}[statement]~
    \begin{verbatim}
// simple statement
// expr statement
; // null statement
// compound statement (block, scope)
// {} empty block is equivalent to ; null statement

// flow-of-control statement
// conditional statements
// if statement
if (cond)
    statement;
else if (cond)
    statement;
else
    statement;

// switch statement
switch (v)
{
    case expr1: // constexpr
        statement;
        break;
    case expr2:
        statement;
        break;
    ...
    default:   // no case match
        statement;
        break;
}

// iterative statement
while(cond)
    statement;

// traditional for statement
for (init-statement; cond; expr) // loop var used locally inside for body
    statement;

// range for statement
// iterate through elems of a container or sequence
for (declaration : sequence) // ref to write elem
    statement;

// do while statement
do
    statement
while (cond);

// jump statements:
// break statement
break; // terminate nearest loop or switch

// continue statement
// terminate current iteration of nearest loop and begin next iteration
continue;

// goto statement
label:
    ...
goto label;

// statement scope
// vars defined in control structure are only visible within scope
\end{verbatim}
\end{definition}

\begin{definition}[class]~
    Data abstraction and encapsulation: separation of interface and implementation
    \begin{itemize}
        \item interface: operations that users can execute
        \item implementation: class data members and funcs
    \end{itemize}

    \begin{verbatim}
// header
class MyClass
{
    friend type nonmember_func(params);
    // allow nonmember func to access class nonpublic members
    // friends are not affected by access control
    // group friend decls at beginning or end of class

public:
    // member funcs access object through implicit parameter this
    // any use of member of class inside member func is an implicit reference through this
    // const member func
    type func(params) const; // const indicates that this is a pointer to const
}

// declare friend in class header
type nonmember_func(params);

// source
type MyClass::func(params) const {...}
type nonmember_func(params) {...}

// main
MyClass myObj;
\end{verbatim}
\end{definition}

\subsection{Additional Class Features}
\begin{definition}[type member]~
    \begin{verbatim}
class MyClass
{
    typedef type T; // type member must appear before they are used
    // appear at beginning of class

    T var;
}
\end{verbatim}
\end{definition}

\begin{definition}[inline member func]~
    \begin{verbatim}
// member funcs defined inside class are automatically inline
// member funcs defined outside class need to be specified inline
class MyClass
{
    type func(params);
}

inline type func(params) {...}
// inline member func should be defined in class header
\end{verbatim}
\end{definition}

\begin{definition}[mutable data member]
    Never const even when it is member of const object.
    \begin{verbatim}
mutable type var;
\end{verbatim}
\end{definition}

\subsection{Constructor}
\begin{definition}[constructor]
    Class initialization, copy, move, assignment, destruction.
    \begin{verbatim}
class MyClass
{
public:
    // constructor funcs
    // initialize data members
    // same name as class
    // no return type

    // default constructor
    // direct initialization
    // automatically called when an obj is created
    MyClass() {}
    MyClass(params) member(expr), ... {...}

    // copy constructor
    // copy initialization
    // copy each nonstatic mem from given object into one being created
    // first param is ref to class
    MyClass(const MyClass &);

private:
    // in-class initializer
    // = for copy initialization
    // braces for direct initialization
}
\end{verbatim}
\end{definition}
