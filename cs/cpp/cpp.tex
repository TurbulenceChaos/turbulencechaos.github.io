\subsection{Qualifier}
\begin{definition}[const]
    The compiler will replace the const var with its value.
    \begin{verbatim}
// const must be initialized
const int i = 1;

// const var is local to file by default
// must specify extern to be used in other files
// source.cpp
extern const i = 10;
// header.h
extern const i;
// main.cpp
#include "header.h"
i ...

// ref to const
// cannot change the object to which it bounds
const int i = 1;
const int &r = i;
const int &r2 = 2; // compiler: const int temp = 2; const int &r2 = temp;

// pointer to const
// cannot change the object to which it points
const int i = 1;
const int *p = &i;

// const pointer
// the address it holds cannot be changed
int i = 1;
int *const p = &i; // p always points to i
\end{verbatim}
\end{definition}

\begin{definition}[constant expr]
    A constant expression is an expr whose value cannot change at compile time.
    \begin{verbatim}
constexpr int i = size();   // ok only if size is a constexpr function
constexpr int *p = nullptr; // const pointer, constexpr applies to p
\end{verbatim}
\end{definition}

\begin{definition}[type alias]
    A synonym for a type.
    \begin{verbatim}
typedef int base;
using base = int;
typedef base *p;  // int*: pointer to int
p p1 = &i;

// const p: const modify the given type: int *const: constant pointer to int
const p p2 = 0; // int *const p2 = 0
const p *p3;    // int *const *p3
\end{verbatim}
\end{definition}

\begin{definition}[const level]~
    \begin{verbatim}
// top-level const: an object itself is const, i.e. const pointer
const int i = 1;
int *const p1 = &i;

// low-level const: point or refer to const object
const int *p2 = &i;
const int &r = i;

// when copying, top-level const is ignored,
// low-level const is never ignored
const int *const p3 = &i;
p2 = p3;
\end{verbatim}
\end{definition}

\section{Exprs}
\begin{definition}[expression]
    Composed of operators and operands.
    \begin{verbatim}
// unary operator: act one one operand
// & address-of
// * dereference

// binary operator: act on two operands1
// == equality
// * multiplication

// overloaded operator
// define operator in class

// lvalue and rvalue
// lvalue: can be assignment
// rvalue: cannot be assignment
// lvalue = ...                   return left-hand operand as lvalue
// address-of &lvalue             return a pointer to lvalue as rvalue
// deref * and subscript operator return lvalue
// ++iterator or --iterator       return lvalue
// decltype(lvalue)               return ref
// decltype(rvalue)               return the exact type of rvalue
decltype(*p) // int&
decltype(&i) // int*
decltype(&p) // int**

// compound expr with two or more operators
// precedence:
// operands of operators with higher precedence group tightly than
// operands of operators at lower precedence
// associativity: group operands with same precedence
// left associative: operators with same precedence group from left-most to right
// right associative: operators with same precedence group from right-most to left
// () force grouping

// arithmetic operator
// left associative
// return rvalue
// reminder: int % int
// return same sign as m
(-m) % n // -(m % n)
m % (-n) // -(m % n)

// logical and relational operator
// return bool val
// 0 return false
// non-zero return true
// guarantee eval order
expr1 && expr2 // eval expr2 if and only if expr1 is true
expr1 || expr2 // eval expr2 if and only if expr1 is false
// precedence:
// right associative
// 1. !
// left associative
// 2. <=, >=
// 3. ==, !=
// 4. &&
// 5. ||
i != j < k // i != (j < k)

// assignment operator
// low precedence
// right associative
// left-hand operand: a modifiable lvalue
// return left-hand operand lvalue
// compound assignment operator
// *=, /=, %=
a *= b;    // compound assignment: a eval only once
a = a * b; // ordinary assignment: a eval twice

// increment and decrement operator
// require lvalue operand
// prefix:  increment operand and return the changed object as lvalue
// postfix: increment operand and return object's original value as rvalue
// use postfix only when necessary
// precedence: ++-- > dereference
*b++ // *(b++)

// evaluation order

// member access operator
p->mem => (*p).mem
// precedence: . > dereference
pointer-> return lvalue
.mem return lvalue or rvalue same as mem

// conditional operator
cond ? expr1 : expr2;
// return lvalue if both expr1 and expr2 are lvalue
// otherwise return rvalue
cond ? expr1 : cond ? expr2 : expr3
// right associative
// low precedence, need ()

// bitwise operator
// hardware

// shift operator (IO operator)
// left associative
// precedence:
1. arithmetic operator
2. IO operator
3. relational, assignment, conditional operator

// sizeof operator
// right associative
// return constexpr size_t in byte
// 1 byte = 8 bits
sizeof(type)
sizeof(expr)
// array
sizeof(a) // sizeof(*a) * size

// comma operator
// guarantee eval order from left to right
// lowest precedence
// expr1, expr2
// return expr2 lvalue or rvalue
for (int v1=..., v2=...; cond; expr1, expr2)
    ...
\end{verbatim}
\end{definition}

    statement

    ...

\section{Function}
\begin{definition}[func]
    Block of code with name.
    \begin{verbatim}
type func(params) // each param with single declarator: int v1, int v2, ...
{
    statement;
    return r;
}

// automatic objects
// params destroyed after function call

// static object
// lifetime continues after function call
static type v = ...;

// func decl
// a function must be declared before use
// semicolon replace the func body in declaration
// func.h

// func.cpp
// header that decls func should be included in source file that defines func
#include "func.h"
type func(params) {...}
// main.cpp
#include "func.h"
func(arguments);

// separate compilation
// g++ -c compile and assemble but do not link, generate object file
g++ -c main.cpp   // main.o
g++ -c func.cpp   // func.o
g++ main.o func.o // link object files, generate a.out

// param and argument
// param are initialized in the same way that vars are initialized:
// type param = argument;
param is ref     => called by ref, param is an alias for argument to which it is bound
param is not ref => called by value, argument value is copied
// pointer param
// copy pointer value
void reset(int* p)
{
    *p = 2; // change value of object to which p point
    p = 0;  // only change the local copy of p, argument is unchanged
}

int i = 1;
reset(&i); // i = 2

// C use pointer param
// C++ often use reference param instead
// ref param
void reset(int& r)
{
    r = 2;
}

int i = 1;
reset(i); // i = 2

// ref param for large containers
void func(const std::string &r);
// ref param for multiple results return

// const param
// argument -> param: top-level const of argument is ignored
// argument -> pointer or ref param: low-level const of argument cannot be ignored
// when a func do not need write access to argument object,
// use ref or pointer to const param for std::string, std::vector, and array

// array param
// cannot copy array
// array converted to a pointer to the first elem in array
void func(const int*);
// size of array
// C-style char string: stop when seeing null char
void func(const char*);
// std lib
void func(const int *b, const int *e);
// define a param that indicates the size of array
void func(const int *, size_t size);

// array ref param
void func(int (&a)[dim]); // param is ref to array of n ints

// multidimensional array argument

void print(int (*matrix)[dim], int rows, ...)

// varying number of params
// std::initializer_list: template type
void func(params, std::initializer_list<T> il);
// number of arguments of a single type in il
// elems in il are always const
void error_msg(std::initializer_list<std::string> il)
{
    for (auto b = il.begin(); b != il.end(); ++b)

}

// ellipsis params
