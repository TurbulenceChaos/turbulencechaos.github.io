\subsection{Compound Types}
\begin{definition}[compound type]
    Compound type is defined in terms of another type.
\end{definition}

\begin{definition}[reference]
    A reference defines an alternative name for an object.
    \begin{verbatim}
// reference must be initialized with object
type var = ...;
type &ref = var; // bind reference to its initializer
// reference cannot be rebind
\end{verbatim}
\end{definition}

\begin{definition}[pointer]
    A pointer is an object that holds the address of another object.
    \begin{verbatim}
type var;
type *p = &var; // pointer to var
// pointer type and object type to which pointer points must match
*p // dereference return lvalue object to which pointer points
p = nullptr; p = 0; // null pointer doesn't pointer to any object

// void* pointer
void *p = &var; // can hold address of any object type
// cannot use void* to operate on object it address
// use void* pointer to deal with memory rather than access object stored in that memory

// pointer to pointer
type **pp = &p; // indicate each pointer level by its own *

// reference to pointer
type *&ref = p; // refer to type*
\end{verbatim}
\end{definition}

\begin{definition}[compound type declaration]~
    \begin{verbatim}
// variable definition consists of a base type and a list of declarators
// each declarator can relate its variable to the base type
type var = ..., *p = &var, &ref = var;
\end{verbatim}
\end{definition}

\subsection{const}
\begin{definition}[const]
    const makes a variable unchangeable with compiler replacing const variable with its value.
    \begin{verbatim}
// const object must be initialized at compile time or run time
// use an object to initialize another object,
// it doesn't matter whether either or both of objects are consts
// copy an object doesn't change that object
const type var = ...;
\end{verbatim}
\end{definition}

\begin{definition}[const objects are local to file]~
    \begin{verbatim}
// const variables are local to file by default
// to share a const object among multiple files,
// use extern on both its definition and declaration
// header
extern const type var; // extern signifies that var is not local to this file,
// and that its definition will occur elsewhere
// source
#include "header.h"
extern const type var = ...;
// main
#include "header.h"
var
\end{verbatim}
\end{definition}

\begin{definition}[const reference]~
    \begin{verbatim}
// bind reference to const to const object
// cannot be used to change the object to which reference bounds
const type var = ...;
const type &ref = var;
// can initialize from any expression that can be converted to the type of reference,
// i.e. can bind const reference to nonconst object, literal, or expression
double d = 3.14;
const int &ref = d; // const int temp = d; const int &ref = temp;
// bound to an unnamed temporary object created by compiler when it needs place
// to store result from expression
\end{verbatim}
\end{definition}

\begin{definition}[pointer to const]~
    \begin{verbatim}
// cannot change the object to which the pointer points
const type var = ...;
const type *p = &var;
// can point to nonconst object
// pointer to const only affects what we can do with pointer
\end{verbatim}
\end{definition}

\begin{definition}[const pointer]~
    \begin{verbatim}
// const pointer must be initialized
// the address pointer holds cannot be changed
type var = ...;
type *const p = &var; // p always points to var
// const pointer says nothing about whether we can use pointer to change object,
// which depends on type to which pointer points
\end{verbatim}
\end{definition}

\begin{definition}[const level]~
    \begin{verbatim}
// top-level const: an object itself is const and cannot be changed
const type var = ...;
int *const p = &var;

// low-level const: point or refer to a const object
// const reference is always low-level
const type *p = &var;
const type &ref = var;

// top-level const is ignored when copying an object
// low-level const is never ignored
const type *const p1 = &var;
const type *p2 = p1;
\end{verbatim}
\end{definition}

\begin{definition}[constexpr]
    The value of constant expression cannot change and can be evaluated at compile time (not until run time).
    \begin{verbatim}
// whether an object is constant expression depends on types and initializers
// literal and const object that is initialized
// from constant expression is constant expression
const type var = ...;
// declare variable in constexpr declaration to ask compiler
// to verify that variable is a constant expression
// variable declared as constexpr are implicitly const and
// must be initialized by constant expression
constexpr type var = func(arguments); // can use constexpr function in the initializer

// can only use constexpr on literal types
// arithmetic, reference, pointer types are literal types
// constexpr pointer can be initialized from nullptr or 0 literal, or
// point/bind to fixed address object
// address of object defined outside any function is constant expression

// constexpr applies to pointer, not the type to which pointer points
// constexpr imposes top-level const on the object it defines
const type *p = nullptr;       // pointer to const type
constexpr type *p = nullptr;   // const pointer to type
constexpr const type *p = ...; // const pointer to const type
\end{verbatim}
\end{definition}

\begin{definition}[typedef]
    A type alias is a name that is synonym for another type.
    \begin{verbatim}
typedef type T;
// alias declaration
using T = type;

typedef type *T; // type* pointer to type
// const appears in the base type modify the given type
// const T is const pointer to type
const T p = 0; // const pointer to type
const T *p;    // pointer to const pointer to type
\end{verbatim}
\end{definition}

\begin{definition}[auto]
    Use auto type specifier to let compiler to deduce type from initializer.
    \begin{verbatim}
// variable that uses auto specifier must have an initializer
auto a = ...;

// when using reference as initializer,
// the initializer is the corresponding object,
// and the compiler uses that object's type for auto's type deduction
// auto drops top-level consts and low-level consts are kept
const type var = ...;
const type &ref = var;
auto a = ref; // initializer object is const type var

// make deduced type have top-level const
const auto a = var; // a is const type

// reference to auto-deduced type,
// where top-level consts in initializer are not ignored
auto &a = var; // a is const type bound to var
\end{verbatim}
\end{definition}

\begin{definition}[decltype]
    decltype type specifier returns type of its operand without evaluating the expression.
    \begin{verbatim}
decltype(expr) d = ...;
// decltype returns the type of variable, including top-level const and references
const type var = ..., &ref = var, *p = &var;
decltype(var) d = ...; // d is const type
decltype(ref) d = ...; // d is const type& and must be initialized
// decltype return a reference type for expressions that yield lvalue objects
// type deduced by dereference a pointer is reference and must be initialized
decltype(*p) d = ...; // d is const type& reference

// decltype of a parenthesized variable is always a reference
decltype((var)) d = ...; // d is const type& and must be initialized
\end{verbatim}
\end{definition}

#include <string>
// using std::string;

// class defines how objects can be initialized
std::string s = "A";   // not include null char at the end of string literal
std::string s(N, 'c'); // N copies of char

// read string
std::cin >> s1 >> s2;

// read line
std::getline(std::cin, s);

// compare
// at least one should be std::string
std::string s = "Hello";
s < "Hiya" // compare first diff char ASCII: i > e
s < "Hello World"

// concat
// cant add two string literals, i.e. "A" + "B"
s + " World\n" // "Hello World\n"

// for range statement: iterate through elems in a given sequence
for (char c : s)
    ...
for (char &r : s) // change char in string
    r = toupper(r); // uppercase

// ispunct(c): check if char is punctuation, i.e. ',', '!', ...
// isspace(c): check if char is space

// subscript operator: s[n]
// !s.empty()
// s.size(): unsigned std::string::size_type
// s.size() < -1 => 2^32 - 1 for 32-bit int
// n: unsigned std::string::size_type
// 0 <= n < s.size()
for(decltype(s.size()) i = 0; i != s.size(); i = i + 1)
    ...

// iterator
\end{verbatim}
\end{definition}

\subsection{Arrays of Arrays}

constexpr size_t rows=..., cols=...;
int a[rows][cols] = {0};

// subscript operator

// for range statement
// control vars except innermost array must be ref
for (auto &row : a)     // for every elem in outer array
    for(auto col : row) // for every elem in inner array

// typedef int int_array[rows];
for (int_array *p = a; p != a + rows; ++p)
    ...
\end{verbatim}
\end{definition}

\section{Exprs}
\begin{definition}[expression]
    Composed of operators and operands.
    \begin{verbatim}
// unary operator: act one one operand
// & address-of
// * dereference

// binary operator: act on two operands1
// == equality
// * multiplication

// overloaded operator
// define operator in class

// lvalue and rvalue
// lvalue: can be assignment
// rvalue: cannot be assignment
// lvalue = ...                   return left-hand operand as lvalue
// address-of &lvalue             return a pointer to lvalue as rvalue
// deref * and subscript operator return lvalue
// ++iterator or --iterator       return lvalue
// decltype(lvalue)               return ref
// decltype(rvalue)               return the exact type of rvalue
decltype(*p) // int&
decltype(&i) // int*
decltype(&p) // int**

// compound expr with two or more operators
// precedence:
// operands of operators with higher precedence group tightly than
// operands of operators at lower precedence
// associativity: group operands with same precedence
// left associative: operators with same precedence group from left-most to right
// right associative: operators with same precedence group from right-most to left
// () force grouping

// arithmetic operator
// left associative
// return rvalue
// reminder: int % int
// return same sign as m
(-m) % n // -(m % n)
m % (-n) // -(m % n)

// logical and relational operator
// return bool val
// 0 return false
// non-zero return true
// guarantee eval order
expr1 && expr2 // eval expr2 if and only if expr1 is true
expr1 || expr2 // eval expr2 if and only if expr1 is false
// precedence:
// right associative
// 1. !
// left associative
// 2. <=, >=
// 3. ==, !=
// 4. &&
// 5. ||
i != j < k // i != (j < k)

// assignment operator
// low precedence
// right associative
// left-hand operand: a modifiable lvalue
// return left-hand operand lvalue
// compound assignment operator
// *=, /=, %=
a *= b;    // compound assignment: a eval only once
a = a * b; // ordinary assignment: a eval twice

// increment and decrement operator
// require lvalue operand
// prefix:  increment operand and return the changed object as lvalue
// postfix: increment operand and return object's original value as rvalue
// use postfix only when necessary
// precedence: ++-- > dereference
*b++ // *(b++)

// evaluation order

// member access operator
p->mem => (*p).mem
// precedence: . > dereference
pointer-> return lvalue
.mem return lvalue or rvalue same as mem

// conditional operator
cond ? expr1 : expr2;
// return lvalue if both expr1 and expr2 are lvalue
// otherwise return rvalue
cond ? expr1 : cond ? expr2 : expr3
// right associative
// low precedence, need ()

// bitwise operator
// hardware

// shift operator (IO operator)
// left associative
// precedence:
1. arithmetic operator
2. IO operator
3. relational, assignment, conditional operator

// sizeof operator
// right associative
// return constexpr size_t in byte
// 1 byte = 8 bits
sizeof(type)
sizeof(expr)
// array
sizeof(a) // sizeof(*a) * size

// comma operator
// guarantee eval order from left to right
// lowest precedence
// expr1, expr2
// return expr2 lvalue or rvalue
for (int v1=..., v2=...; cond; expr1, expr2)
    ...
\end{verbatim}
\end{definition}

\begin{definition}[IO objects]~
    \begin{verbatim}
#include <iostream>
// using std::cout, std::cin;
std::cout << "str"; // return left operand: std::cout
std::cin  >> v;     // return left operand: std::cin

// istream become invalid when
// 1. reading a non-integer val: 1 2 3.1
// 2. end-of-file: Ctrl-z RET (Windows), Ctrl-d RET (Linux)
\end{verbatim}
\end{definition}

\section{Statement}
\begin{definition}[statement]~
    \begin{verbatim}
// simple statement
// expr statement
; // null statement
// compound statement (block, scope)
// {} empty block is equivalent to ; null statement

// flow-of-control statement
// conditional statements
// if statement
if (cond)
    statement;
else if (cond)
    statement;
else
    statement;

// switch statement
switch (v)
{
    case expr1: // constexpr
        statement;
        break;
    case expr2:
        statement;
        break;
    ...
    default:   // no case match
        statement;
        break;
}

// iterative statement
while(cond)
    statement;

// traditional for statement
for (init-statement; cond; expr) // loop var used locally inside for body
    statement;

// range for statement
// iterate through elems of a container or sequence
for (declaration : sequence) // ref to write elem
    statement;

// do while statement
do
    statement
while (cond);

// jump statements:
// break statement
break; // terminate nearest loop or switch

// continue statement
// terminate current iteration of nearest loop and begin next iteration
continue;

// goto statement
label:
    ...
goto label;

// statement scope
// vars defined in control structure are only visible within scope
\end{verbatim}
\end{definition}

\section{Function}

}

func(arguments);

\subsection{Func Declaration}

\subsection{Parameter Initialization}
\begin{definition}[param initialization]~
    \begin{verbatim}
// C++ use reference param to avoid copies
type& param = argument // param bound to argument

// nonreference
type param = argument  // argument value is copied
// C use pointer param to access object
type* param = &argument

// const param
// top-level const are ignored
type param = const argument
// low-level const cannot be ignored
const type& param = const argument
const type* param = &const argument
\end{verbatim}
\end{definition}

\begin{definition}[array param]~
    \begin{verbatim}
// array param
// cannot copy array
type array[N] = {};
type func(const type* param);   // array converted to &array[0]
type func(const type param[]);  // equivalent to const type* param
type func(const type param[M]); // compiler ignores first dimension
// cannot use type var[] = array in expr!

// when func doesn't need write access to array elements,
// array param should be pointer to const

// array size rely on additional inform:
// C-style string stops at null char
type func(const char* cp) {if(cp) ...}
// beg and end pointer to first and last element in array
const int* beg = std::begin(array)
const int* end = std::end(array)
// define a second param indicates array size
type func(const int[], size_t);

// array ref param
int (&param)[N] = array // type size limits usefulness of func

// multidimensional array param
// the second and subsequent dimensions must be specified
type func(int (*matrix)[Y][Z][...], int X);
type func(int matrix[][Y][Z][...], int X); // compiler ignores first dimension

// varying number of params
// std::initializer_list: template type
void func(params, std::initializer_list<T> il);
// number of arguments of a single type in il
// elems in il are always const
void error_msg(std::initializer_list<std::string> il)
{
        for (auto b = il.begin(); b != il.end(); ++b)
        std::cout << *b << " ";
    }

error_msg({"FUNC", s1, s2});

// ellipsis params
void func(params, ...)
\end{verbatim}
\end{definition}

\subsection{Return Statement}
\begin{definition}[return]
    Terminate func and return control to func call.
    \begin{verbatim}
// return statement
// analogous to break statement
return;       // return with no value used in void func(params);
return value; // type temp = value
// return value is used to initialize a temporary as result of func call

// return reference
// funcs that return reference are lvalues
// other return types are rvalues

// call operator
// same precedence as dot and arrow operator
// call operator is left associative
func(argument).member_func() // return class type object

// return list
std::vector<type> temp = { ... }

// cannot return array or func
// return pointer or reference to array or func
type (*func(params)) [N]; // return pointer to array of N type elements

// use typedef
typedef int arrayT[N];    // synonym to array of n ints
using arrayT = int[N];
arrayT* func(params);
// use decltype
int a[] = {1,2,3};
decltype(a) *func(params);
// use trailing return type
auto func(params) -> int(*)[N];

// main return
// 0 indicates success
// nonzero has machine-dependent meaning

PF p;

F *func(parmas);

\begin{definition}[class]~
    Data abstraction and encapsulation: separation of interface and implementation
    \begin{itemize}
        \item interface: operations that users can execute
        \item implementation: class data members and funcs
    \end{itemize}

    \begin{verbatim}

// class MyClass; // forward declaration if class appears in friend func params or return

// declare friend func in class header
// friend func can be defined inside class
// friend func decl must still be provided outside class before other member func use it
type nonmember_func(params);

class MyClass
{
public:
    friend type nonmember_func(params);
    // allow nonmember func to access class nonpublic members
    // friends are not affected by access control
    // group friend decls at beginning or end of class
    // friend class MyClass2; // access all member funcs in MyClass2
    // friend type MyClass2::member_func(params);

    // member funcs access object through implicit parameter this
    // any use of member of class inside member func is an implicit reference through this
    type func(params) const; // const member func indicates that this is a pointer to const
    // const member func cannot change data members of object on which it is called

private:
    // class is not defined until its class body is complete
    // class cannot have data members of its own type
    // class can have data members that are pointers or references to its own type
    type var;
}; // must end with ;

// source
type MyClass::func(params) const { ... }
type nonmember_func(params) { ... }

// main
MyClass myObj;
\end{verbatim}

\begin{definition}[static class members]
    A member is associated with the class rather than with individual obejct of class type
    by adding the keyword \verb|static| to its declaration.
    \begin{verbatim}

// they are not initialized by class constructors when creating objects of the class
// static member functions are not bound to any object and do not have this pointer

static type func(params) { ... }

// we must define and initialize each static data member outside the Class
// static member are defined outside any function like global objects
type MyClass::var = ...;

// static keyword is used only on the declaration inside class body
// static member defined outside the class do not need to repeat the static keyword

// in-class initialization of static data members
// can provide in-class initializer for static members that have integral type
// must do so for static members that are constexpr of literal type
// initializer must be constant expression
static constexpr type var = ...;
// should be defined outside class definition even if
// const static data member is initialized inside class body
constexpr type MyClass::var; // initializer provided in class definition

// can access a static member directly through scope operator
MyClass::func()

// static data member can have incomplete type,
// i.e. same type as the class type
static MyClass var;
\end{verbatim}
\end{definition}

\begin{definition}[constructor initializer]~
    \begin{verbatim}

// automatically called when an obj is created

// provide a default constructor if other constructors are defined

// use default constructor to define obj
MyClass myClass; // error MyClass myClass(); to define func

// constructor initializer list specifies only the value used to initialize members
// members are initialized in the order in which they appear in class definition
// write constructor initializer in same order as members are declared
// avoid using members to initialize other members

// delegating constructor
// use other constructor to perform its initialization
MyClass(params) : var1(expr), ... {} // nondelegating constructor
// delegated-to constructor initializer list and func body are both executed
MyClass() : MyClass(arguments) {} // default constructor
MyClass(params) : MyClass() { ... } // delegate to default constructor
\end{verbatim}
\end{definition}

\begin{definition}[converting constructor]
    A constructor that can be called with a single argument defines implicit converison from
    constructor's parameter type to class type.

Myclass myClass = argument; // copy initialization

// explicit constructor
// prevent implicit conversion by declaring constructor as explicit inside class
// be meaningful only on constructors that can be called with a single argument
// only direct initalization is allowed
explicit MyClass(param) {}

// explicit conversion
MyClass(argument);
std::string("str");
\end{verbatim}
\end{definition}
