\begin{axiom}
  \label{axiom:peano-axioms}
  Peano axioms:

  \begin{enumerate}
  \item 0 is a natural number.
  \item If $n$ is a natural number, then $n\pp$ is also a natural number.
  \item $n\pp \neq 0$
  \item $n \neq m \Rightarrow n\pp \neq m\pp$

    $n\pp = m\pp \Rightarrow n = m$
  \item Principle of mathematical induction:

  Suppose that $P(0)$ is true, and suppose that whenever $P(n)$ is true, $P(n\pp)$ is also true. Then $P(n)$ is true for every natural number $n$.
  \end{enumerate}
\end{axiom}

\begin{definition}
  Natural number system $\mathbb{N}$:

  \begin{gather*}
    1 := 0\pp \\
    2 := 1\pp \\
    3 := 2\pp \\
    \vdots
  \end{gather*}
\end{definition}

\begin{definition}
  Addition:

  Let $m$ be a natural number.

  \begin{align*}
    0 + m & := m \\
    (n\pp) + m & := (n+m)\pp
  \end{align*}
\end{definition}

\begin{lemma}
  For any natural number $n$, $n + 0 = n$.
\end{lemma}

\begin{proof}
  $0 + 0 = 0$
  
  Suppose $n + 0 = n$

  $(n\pp) + 0 = (n+0)\pp = n\pp$
\end{proof}
