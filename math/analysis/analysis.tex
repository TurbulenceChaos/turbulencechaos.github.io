\subsection{Natural Number system}
\begin{itemize}
\item This is how mathematics works - it treats its objects abstractly, caring only about what properties the objects have, not what the objects are or what they mean \cite{TAO_2022_ANALYSIS_I}.
\item It is pointless, at least from a mathematician's standpoint, as to argue about which model is the true one; as long as it obeys all the axioms and does all the right things, that's good enough to do maths.

\subsubsection{The Peano Axioms}
\begin{axiom}
  \label{axiom:peano-axioms}
  Peano axioms:
  \begin{enumerate}
  \item 0 is a natural number.
  \item If $n$ is a natural number, then $n\pp$ is also a natural number.
  \item $n\pp \neq 0$
  \item $n \neq m \Rightarrow n\pp \neq m\pp$

    $n\pp = m\pp \Rightarrow n = m$
  \item Principle of mathematical induction:

    Suppose that $P(0)$ is true, and suppose that whenever $P(n)$ is true, $P(n\pp)$ is also true. Then $P(n)$ is true for every natural number $n$.
  \end{enumerate}
\end{axiom}

\begin{definition}
  \begin{gather*}
    1 := 0\pp \\
    2 := 1\pp \\
    3 := 2\pp \\
    \vdots
  \end{gather*}
\end{definition}

\subsubsection{Addition}
\begin{definition}
  Addition.
  
  Let $n$ and $m$ be natural numbers.
  \begin{enumerate}
  \item $0+m:=m$
  \item $(n\pp)+m:=(n+m)\pp$
  \end{enumerate}
\end{definition}

\begin{lemma}
  Let $n$ and $m$ be natural numbers.
  \begin{enumerate}
  \item $n+0=n$
  \item $n+(m\pp)=(n+m)\pp$
  \end{enumerate}
\end{lemma}

\begin{proposition}
  Let $a$, $b$, $c$ be natural numbers.
  \begin{enumerate}
  \item $a+b=b+a$
  \item $(a+b)+c=a+(b+c)$
  \item $a+b=a+c \Rightarrow b=c$
  \end{enumerate}
\end{proposition}

\begin{definition}
  A natural number $n$ is positive iff (if and only if) $n \neq 0$.
\end{definition}

\begin{proposition}
  \label{prop:positive-natural-numbers}
  Let $a$ and $b$ be natural numbers.
  \begin{enumerate}
  \item If $a$ is positive, then $a+b$ is positive.
  \item $a+b=0 \Rightarrow a=b=0$.
  \item If $a$ is positive, then there exists only one natural number $b$ such that $b\pp=a$.
  \end{enumerate}
\end{proposition}

\begin{proof}
  Prop. \ref{prop:positive-natural-numbers} (1)
  
  $a+0=a$ is positive.
  Suppose that $a+b$ is positive.
  Then $a+b\pp=(a+b)\pp \neq 0$ is positive.
\end{proof}

\begin{proof}
  Prop. \ref{prop:positive-natural-numbers} (3)

  Suppose that $a=b\pp$.
  Then $a\pp=(a)\pp$.
\end{proof}

\begin{definition}
  Let $n$ and $m$ be natural numbers.
  \begin{enumerate}
  \item $n \geq m$ or $m \leq n$ iff $n=m+a$
  \item $n>m$ or $m<n$ iff $n \geq m$ and $n \neq m$
  \end{enumerate}
\end{definition}

\begin{proposition}
  \label{prop:natural-numbers-order}
  Let $a$, $b$, $c$ be natural numbers.
  \begin{enumerate}
  \item $a \geq a$
  \item If $a \geq b$ and $b \geq c$, then $a \geq c$
  \item If $a \geq b$ and $b \geq a$, then $a=b$
  \item $a \geq b$ iff $a+c \geq b+c$
  \item $a<b$ iff $a\pp \leq b$
  \item $a<b$ iff $b=a+d$ for some positive number $d$
  \end{enumerate}
\end{proposition}

\begin{proof}
  Prop. \ref{prop:natural-numbers-order} (5)

  $b=a+d$ and $b \neq a \Rightarrow d \neq 0 \Rightarrow d = n\pp$

  $b=a+(n\pp)=(a\pp)+n \Rightarrow a\pp \leq b$
\end{proof}

\begin{proposition}
  Let $a$ and $b$ be natural numbers. Then exactly one of following statement is true:
  $a<b$, $a=b$, or $a>b$.
\end{proposition}

\begin{proof}
  If $a<b$ and $a>b \overset{\text{Prop. \ref{prop:natural-numbers-order} (3)}}{\Rightarrow} a=b$ contradiction
\end{proof}

\begin{proposition}
  Strong principle of induction.

  Let $m_{0}$ be a natural number. Suppose that for each $m \geq m_{0}$, if $P(m')$ is true for all $m_{0} \leq m' < m$, then $P(m)$ is also true. Then $P(m)$ is true for all $m \geq m_{0}$.
\end{proposition}

\begin{proof}
  Let $Q(n)$ be the property that $P(m)$ is true for all $m_{0} \leq m < n$.

  Suppose that if $Q(n)$ is true, then $P(m)$ is also true.

  Then $P(m)$ is true for all $m_{0} \leq m < n\pp$.
\end{proof}

\begin{example}
  Fibonacci numbers.
  \begin{gather}
    \label{eq:fibonacci-numbers}
    F_{n} = \frac{\phi^{n} - \psi^{n}}{\phi-\psi} \\
    F_{n+2} = F_{n+1} + F_{n}
  \end{gather}
  where $\phi=\frac{1+\sqrt{5}}{2}$ and $\psi=\frac{1-\sqrt{5}}{2}$
\end{example}

\begin{exercise}
  Principle of backwards induction.

  Let $n$ be a natural number. Whenever $P(m\pp)$ is true, $P(m)$ is true.
  If $P(n)$ is true, then $P(m)$ is true for all $m \leq n$.
\end{exercise}

\begin{proof}
  Suppose that if $P(n)$ is true, then $P(m)$ is true for all $m \leq n$.
  If $P(n\pp)$ is true, then $P(m)$ is true for all $m \leq n\pp$.
\end{proof}

\begin{exercise}
  Principle of induction starting from the base case $n$.

  Let $n$ be a natural number. Whenever $P(m)$ is true, $P(m\pp)$ is true.
  If $P(n)$ is true, then $P(m)$ is true for all $m \geq n$.
\end{exercise}

\begin{proof}
  If $P(0)$ is true, suppose that $P(m)$ is true.
  Then $P(m\pp)$ is also true.
  Then $P(m)$ is true for all $m \geq 0$.
  Suppose that if $P(n)$ is true, then $P(m)$ is true for all $m \geq n$.
  If $P(n\pp)$ is true, then $P(m)$ is true for all $m \geq n\pp$.
\end{proof}

\subsubsection{Multiplication}
\begin{definition}
  Let $n$ and $m$ be natural numbers.
  \begin{enumerate}
  \item $0 \times m := 0$
  \item $(n\pp) \times m := (n \times m)+m$
  \end{enumerate}
\end{definition}

\begin{lemma}
  Let $n$ and $m$ be natural numbers.
  \begin{enumerate}
  \item $n \times m = m \times n$
  \item $n \times m = 0$ iff at least one of $n$ and $m$ is equal to 0.
  \end{enumerate}
\end{lemma}

\begin{proposition}
  Let $a$, $b$, $c$ be natural numbers.
  \begin{enumerate}
  \item $a(b+c)=(b+c)a=bc+ca$
  \item $(ab)c=a(bc)$
  \item If $a<b$ and $c$ is positive, then $ac<nc$.
  \item If $ac=bc$ and $c$ is positive, then $a=b$.
  \end{enumerate}
\end{proposition}

\begin{lemma}
  Euclid's division lemma.

  Let $n$ be a natural number and $q$ be a positive natural number. Then there exist natural numbers $m$ and $r$, such that $0 \leq r < q$ and $n=mq+r$.
\end{lemma}

\begin{definition}
  Exponentiation.

  \begin{enumerate}
  \item $m^{0}:=1$ and $0^{0}:=1$
  \item $m^{n\pp}:=m^{n} \times m$
  \end{enumerate}
\end{definition}
