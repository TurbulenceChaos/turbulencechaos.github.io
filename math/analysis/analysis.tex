\subsection{Natural Numbers}
\begin{axiom}
  0 is a natural number.
\end{axiom}

\begin{axiom}
  If $n$ is a natural number, then $n++$ is also a natural number. 
\end{axiom}

\begin{definition}
  \begin{gather*}
    1 := 0++ \\
    2 := 1++ \\
    3 := 2++ \\
    \vdots
  \end{gather*}
\end{definition}

\begin{proposition}
  3 is a natural number.
\end{proposition}
