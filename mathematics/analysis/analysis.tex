\section{Natural Number}
\begin{definition}[natural number]
    A natural number is any element of the set
    \begin{equation}
        \mathbb{N} := {0, 1, 2, 3, \dots}
    \end{equation}
\end{definition}

\begin{axiom}[0]
    0 is a natural number.
\end{axiom}

\begin{axiom}[successor]
    If $n$ is a natural number, then $n{++}$ is also a natural number.
\end{axiom}

\begin{definition}[notation]
    $1 := 0{++}$, $2 := 1{++}$, $3 := 2{++}$, $\dots$
\end{definition}

\begin{axiom}[$n{++} \neq 0$]
    0 is not the successor of any natural number $n$.
\end{axiom}

\begin{axiom}[$n{++} \neq m{++}$]
    Different natural numbers must have different successors.

    If $n \neq m$, then $n{++} \neq m{++}$.
    Equivalently, if $n{++} = m{++}$, then $n = m$.
\end{axiom}

\begin{axiom}[principle of mathematical induction]
    Let $P(n)$ be any property pertaining to a natural number $n$.
    Suppose that $P(0)$ is true, and suppose that whenever $P(n)$ is true,
    $P(n{++})$ is also true. Then $P(n)$ is true for every natural number $n$.
\end{axiom}

\begin{lemma}[recursive definition of sequence]
    $a_{0} = c$ and $a_{n{++} = f_{n}(a_{n})}$ for each natural number $n$.
\end{lemma}

\subsection{Addition}
\begin{definition}[addition]
    Define $0+m:=m$. Suppose inductively that we have defined $n+m$.
    Then $(n{++})+m:=(n+m){++}$.
\end{definition}

$a_{n} = n+m$ and $f_{n}(a_{n}) = a_{n}{++}$

\begin{lemma}
    $n+0=n$
\end{lemma}

\begin{lemma}
    $n+(m{++})=(n+m){++}$
\end{lemma}

\begin{proposition}[addition is commutative]
    \begin{equation}
        n+m = m+n
    \end{equation}
\end{proposition}

\begin{proposition}[addition is associative]
    \begin{equation}
        (a+b)+c = a+(b+c)
    \end{equation}
\end{proposition}

\begin{proposition}[cancellation law]
    If $a+b = a+c$, then $b=c$.
\end{proposition}

\begin{definition}
    A natural number $n$ is positive if and only if $n \neq 0$.
\end{definition}

\begin{lemma}
    If $a$ is a positive natural number and $b$ is a natural number,
    then $a+b$ is positive.
\end{lemma}

\begin{lemma}
    If $a+b = 0$, then $a=0$ and $b=0$.
\end{lemma}

\begin{lemma}
    If $a$ is a positive number, then there exists exactly one natural number $b$
    such that $b{++} = a$.
\end{lemma}

\begin{definition}[order]~
    \begin{itemize}
        \item $n \geq m$ or $m \leq n$ if and only if $n=m+a$ for some natural number $a$
        \item $n>m$ or $m<n$ if and only if $n \geq m$ and $n \neq m$
    \end{itemize}
\end{definition}

\begin{proposition}[properties of order]~
    \begin{itemize}
        \item reflexive: $a \geq a$
        \item transitive: If $a \geq b$ and $b \geq c$, then $a \geq c$
        \item antisymmetric: If $a \geq b$ and $b \geq a$, then $a=b$
        \item addition preserve order: $a \geq b$ if and only if $a+c \geq b+c$
        \item $a<b$ if and only if $a{++} \leq b$
        \item $a<b$ if and only if $b=a+d$ for some positive number $d$
    \end{itemize}
\end{proposition}

\begin{proposition}[trichotomy of order]
    Exactly one of following statement is true:
    $a<b$, $a=b$, or $a>b$.
\end{proposition}

\begin{proposition}[strong principle of induction]
    Let $m_{0}$ be a natural number.
    Suppose that for each $m \geq m_{0}$, if $P(m')$ is true for all $m_{0} \leq m' < m$,
    then $P(m)$ is also true. Then $P(m)$ is true for all $m \geq m_{0}$.
\end{proposition}

\begin{example}[Fibonacci sequence]
    \begin{gather}
        F_{0} = 0 \\
        F_{1} = 1 \\
        F_{n} = F_{n-1} + F_{n-2}
    \end{gather}
\end{example}

\subsection{Multiplication}
\begin{definition}[multiplication]
    Define $0 \times m := 0$.
    Suppose inductively that we have defined $n \times m$.
    Then $(n{++}) \times m := (n \times m) + m$.
\end{definition}

\begin{lemma}
    $n \times 0 = 0 \times n$
\end{lemma}

\begin{lemma}
    $n \times (m{++}) = (n \times m) + n$
\end{lemma}

\begin{proposition}[multiplication is commutative]
    \begin{equation}
        n \times m = m \times n
    \end{equation}
\end{proposition}

\begin{lemma}[zero divisor]
    $n \times m = 0$ if and only if at least one of $n$ and $m$ is equal to 0.
    If $n$ and $m$ are both positive, then $nm$ is also positive.
\end{lemma}

\begin{proposition}[distributive law]
    \begin{equation}
        a(b+c)=(b+c)a=bc+ca
    \end{equation}
\end{proposition}

\begin{proposition}[multiplication is associative]
    \begin{equation}
        (ab)c=a(bc)
    \end{equation}
\end{proposition}

\begin{lemma}[multiplication preserve order]
    If $a<b$ and $c$ is positive, then $ac<nc$.
\end{lemma}

\begin{proposition}[cancellation law]
    If $ac=bc$ and $c$ is positive, then $a=b$.
\end{proposition}

\begin{proposition}[Euclid's division lemma]
    Let $n$ be a natural number and $q$ be a positive natural number.
    Then there exist natural numbers $m$ and $r$, such that $0 \leq r < q$ and $n=mq+r$.
\end{proposition}

\begin{proof}
    $0 = 0 \times q + 0$ and $0 \leq r < q$.
    Suppose that $n=mq+r$ and $0 \leq r < q$.
    Then $n{++}=mq+(r{++})$ and $0 \leq r{++} \leq q$.
    If $r{++}=q$, $n{++}=mq+q=(m+1)q+0$.
    If $r{++}<q$, $n{++}=mq+r{++}$.
\end{proof}

\begin{definition}[Exponentiation]
    Define $m^{0}:=1$, in particular $0^{0}:=1$.
    Suppose recursively that $m^{n}$ has been defined for some natural number $n$,
    then define $m^{n{++}}:=m^{n} \times m$.
\end{definition}

\section{Set Theory}
\begin{definition}[set]
    Define a set $A$ to be any unordered collection of objects.
    If $x$ is an element of A, then $x \in A$, otherwise $x \notin A$.
\end{definition}

\begin{axiom}[sets are objects]
    If $A$ is a set, then $A$ is also an object.
\end{axiom}

\begin{axiom}[equality of sets]
    $A=B$ if and only if every element of $A$ is an element of $B$ and vice versa.
\end{axiom}

\begin{axiom}[empty set]
    There exists an empty set $\emptyset$ which contains no elements.
\end{axiom}

\begin{lemma}[single choice]
    If $A$ is a non-empty set, then there exists an object $x$ such that $x \in A$.
\end{lemma}

\begin{axiom}[singleton and pair sets]
    If $a$ is an object, then there exists a singleton set $\{a\}$ whose only element is $a$.
    If $a$ and $b$ are objects, then there exists a pair set $\{a,b\}$ whose only elements are $a$ and $b$.
\end{axiom}

\begin{axiom}[pairwise union]
    If $A$ and $B$ are two sets, then there exists a union of $A$ and $B$ set $A \cup B$,
    which consists of all the elements which belong to $A$ or $B$ or both.
    \begin{equation}
        x \in A \cup B \Longleftrightarrow x \in A \text{ or } x \in B
    \end{equation}
\end{axiom}

\begin{proposition}[properties of union]~
    \begin{itemize}
        \item $\{a,b\} = \{a\} \cup \{b\}$
        \item commutative: $A \cup B = B \cup A$
        \item associative: $(A \cup B) \cup C = A \cup (B \cup C)$
    \end{itemize}
\end{proposition}

\begin{definition}[subsets]
    $A$ is a subset of $B$, denoted $A \subseteq B$,
    if and only if every element of $A$ is also an element of $B$.

    $A$ is a proper subset of $B$, denoted $A \subsetneq B$,
    if $A \subseteq B$ and $A \neq B$.
\end{definition}

\begin{axiom}[axiom of specification]
    Let $A$ be a set and $P(x)$ be a property pertaining to $x$ for each $x \in A$.
    Then there exists a set $\{x \in A : P(x) \text{ is true}\}$
    whose elements are precisely the elements $x$ in $A$ for which $P(x)$ is true.
\end{axiom}

\begin{definition}[intersections]
    The intersection of two sets is defined to be the set
    \begin{equation}
        A \cap B := \{x \in A : x \in B\}
    \end{equation}
\end{definition}

\begin{definition}[difference sets]
    Define $A-B$ or $A \backslash B$ to be set $A$ with elements of $B$ removed
    \begin{equation}
        A \backslash B := \{x \in A : x \notin B\}
    \end{equation}
\end{definition}

\begin{axiom}[replacement]
    If $P(x,y)$ is a property pertaining to $x$ and $y$,
    such that for each $x \in A$ there is at most one $y$ for which $P(x,y)$ is true.
    Then there exists a set $\{y : P(x,y) \text{ is true for some } x \in A\}$.
\end{axiom}

\begin{axiom}[infinity]
    There exists a set $\mathbb{N}$, whose elements are called natural numbers,
    as well as an object 0 in $\mathbb{N}$,
    and an object $n{++}$ assigned to every $n \in \mathbb{N}$,
    such that the Peano axioms hold.
\end{axiom}

\subsection{Cartesian Product}
\begin{definition}[ordered pair]
    If $x$ and $y$ are any objects,
    define the ordered pair $(x,y)$ to be a new object.
    Two ordered pairs $(x,y)$ and $(x',y')$ are equal if and only if both their components match
    \begin{equation}
        (x,y) = (x',y') \Longleftrightarrow \text{$x = x'$ and $y = y'$.}
    \end{equation}
\end{definition}

\begin{definition}[Cartesian product]
    If $X$ and $Y$ are sets, then define Cartesian product $X \times Y$ to be the collection of ordered pair
    \begin{equation}
        X \times Y = \{(x,y): x \in X, y \in Y\}
    \end{equation}
\end{definition}

\begin{definition}[ordered sequence of $n$ elements]
    If $(x_{1},\dots,x_{n})$ is an ordered n-tuple,
    define Cartesian product $X_{1} \times \dots \times X_{n}$ by
    \begin{equation}
        \prod_{1 \leq i \leq n}X_{i} :=
        \{(x_{i})_{1 \leq i \leq n}: x_{i} \in X_{i} \text{ for all } 1 \leq i \leq n\}
    \end{equation}
\end{definition}

\begin{lemma}[finite choice]
    Let $X_{i}$ be a non-empty set, then there exists an n-tuple $(x_{i})_{1 \leq i \leq n}$
    such that $x_{i} \in X_{i}$ for all $1 \leq i \leq n$.
\end{lemma}

\section{Function}
\begin{definition}[function]
    Let $X$, $Y$ be sets, and let $P(x,y)$ be a property pertaining to
    an object $x \in X$ and an object $y \in Y$, such that
    for every $x \in X$, there is exactly only one $y \in Y$ for which
    $P(x,y)$ is true.
    Then the function $f: X \to Y$ is defined by $P$ on the domain $X$ and codomain (range)
    to be the unique object $f(x) \in Y$ for which $P(x,f(x))$ is true.
\end{definition}

\begin{definition}[equality of functions]
    Two functions $f: X \to Y$ and $g: X' \to Y'$ are equal if their domains and codomains agree,
    and $f(x) = g(x)$ for all $x \in X$.
\end{definition}

\begin{definition}[composition]
    Let $f: X \to Y$ and $g: Y \to Z$ be two functions,
    such that the codomain of $f$ is the same set as the domain of $g$.
    Then define composition $g \circ f: X \to Z$ to the function
    \begin{equation}
        (g \circ f)(x) := g(f(x))
    \end{equation}
\end{definition}

\begin{definition}[injective]
    A function is one-to-one if every element in $X$ maps to different element in $Y$
    \begin{equation}
        x \neq x' \Longrightarrow f(x) \neq f(x')
    \end{equation}
    Equivalently,
    \begin{equation}
        f(x) = f(x') \Longrightarrow x = x'
    \end{equation}
\end{definition}

\begin{definition}[surjective]
    A function is onto if every element in $Y$ comes from applying $f$ to some element in $X$.
    \begin{equation}
        \text{For every $y \in Y$, there exists $x \in X$ such that $f(x) = y$.}
    \end{equation}
\end{definition}

\begin{definition}[bijective]
    A function is bijective or invertible if it is both one-to-one and onto.
\end{definition}

\subsection{Images}
\begin{definition}[images of sets]
    If $f: X \to Y$ is a function from $X$ to $Y$, and $S$ is a subset of $X$,
    define $f(S)$ to be the set
    \begin{equation}
        f(S) := \{f(x) : x \in S\}
    \end{equation}
\end{definition}

\begin{definition}[inverse images]
    If $U$ is a subset of $Y$, define the set $f^{-1}(U)$ to be the set
    \begin{equation}
        f^{-1}(U) := \{x \in X : f(x) \in U\}
    \end{equation}
\end{definition}

\begin{axiom}[power set axiom]
    Let $X$ and $Y$ be sets. Then there exists a set $Y^{X}$
    which consists of all the functions from $X$ to $Y$
    \begin{equation}
        f \in Y^{X} \Longleftrightarrow \text{$f$ is a function with domain $X$ and codomain $Y$.}
    \end{equation}
\end{axiom}

\begin{axiom}[union]
    Let $A$ be a set whose elements are sets.
    Then there exists a set $\bigcup A$
    \begin{equation}
        x \in \bigcup A \Longleftrightarrow \text{$x \in S$ for some $S \in A$.}
    \end{equation}
\end{axiom}

\section{Integers}
\begin{definition}[$\mathbb{Z}$]
    An integer is an expression of the form $a \mathbin{\text{---}} b$,
    where $a$ and $b$ are natural numbers.
    Two integers are considered to be equal $a \mathbin{\text{---}} b = c \mathbin{\text{---}} d$
    if and only if $a + d = c + b$.
\end{definition}

\begin{definition}[sum]
    \begin{equation}
        (a \mathbin{\text{---}}  b) + (c \mathbin{\text{---}} d) := (a + c) \mathbin{\text{---}} (b + d)
    \end{equation}
\end{definition}

\begin{definition}[product]
    \begin{equation}
        (a \mathbin{\text{---}}  b) \times (c \mathbin{\text{---}} d) := (ac + bd) \mathbin{\text{---}} (ad + bc)
    \end{equation}
\end{definition}

\begin{lemma}[substitution axiom verified]
    Let $a, b, a', b', c, d$ be natural numbers. If $(a \mathbin{\text{---}} b) = (a' \mathbin{\text{---}} b')$, then
    $(a \mathbin{\text{---}} b) + (c \mathbin{\text{---}} d) = (a' \mathbin{\text{---}} b') + (c \mathbin{\text{---}} d)$
    and
    $(a \mathbin{\text{---}} b) \times (c \mathbin{\text{---}} d) = (a' \mathbin{\text{---}} b') \times (c \mathbin{\text{---}} d)$,
    and also
    $(c \mathbin{\text{---}} d) + (a \mathbin{\text{---}} b) = (c \mathbin{\text{---}} d) + (a' \mathbin{\text{---}} b')$
    and
    $(c \mathbin{\text{---}} d) \times (a \mathbin{\text{---}} b) = (c \mathbin{\text{---}} d) \times (a' \mathbin{\text{---}} b')$.
\end{lemma}

\begin{definition}[negation]
    If $a \mathbin{\text{---}} b$ is an integer, we define the negation $-(a \mathbin{\text{---}} b)$ to be
    $(b \mathbin{\text{---}} a)$.
    In particular if $n = n \mathbin{\text{---}} 0$ is a positive natural number,
    we define its negation $-n = 0 \mathbin{\text{---}} n$.
\end{definition}

\begin{lemma}[trichotomy of integers]
    Let $x$ be an integer. Then exactly one of the following three statement is true:
    (a) $x$ is 0;
    (b) $x$ is equal to a positive natural number $n$;
    (c) $x$ is the negation $-n$ of a positive natural number $n$.
\end{lemma}

\begin{proposition}[commutative ring]
    Let $x, y, z$ be integers, then
    \begin{gather}
        x + y = y + x \\
        (x + y) + z = x + (y + z) \\
        x + 0 = 0 + x = x \\
        x + (-x) = (-x) + x = 0 \\
        x y = y x \\
        (x y) z = x (y z) \\
        x 1 = 1 x = x \\
        x (y + z) = x y + x z \\
        (y + x) x = y x + z x
    \end{gather}
\end{proposition}

\begin{definition}[substraction]
    \begin{equation}
        x - y := x + (-y)
    \end{equation}
\end{definition}

\begin{proposition}[zero divisor]
    Let $a$ and $b$ be integers such that $a b = 0$.
    Then either $a = 0$ or $b = 0$ (or both).
\end{proposition}

\begin{proposition}[cancellation law]
    If $a, b , c$ are integers such that $a c = b c$ and $c$ is non-zero, then $a = b$.
\end{proposition}

\begin{definition}[order]
    Let $n$ and $m$ be integers.
    $n \geq m$ or $m \leq n$ if and only if $n = m + a$ for some natural number $a$.
    $n > m$ or $n < m$ if and only if $n \geq m$ and $n \neq m$.
\end{definition}

\begin{lemma}[properties of order]
    Let $a, b, c$ be integers, then
    \begin{itemize}
        \item $a > b$ if and only if $a - b$ is a positive natural number.
        \item (addition preserves order)
              If $a > b$, then $a + c > b + c$.
        \item (positive multiplication preserves order)
              If $a > b$ and $c$ is positive, then $a c > b c$.
        \item (negation reverses order)
              If $a > b$, then $-a < -b$.
        \item (order is transitive)
              If $a > b$ and $b > c$, then $a > c$.
        \item (order trichotomy)
              Exactly one of the statements $a > b$, $a < b$, or $a = b$ is true.
    \end{itemize}
\end{lemma}

\section{Rationals}
\begin{definition}[$\mathbb{Q}$]
    A rational number is an expression of the form $a // b$,
    where $a$ and $b$ are integers and $b \neq 0$.
    $a // 0$ is not considered to be a rational number.
    $a // b = c // d$ if and only if $a d = c b$.
\end{definition}

\begin{definition}[sum]
    \begin{equation}
        (a // b) + (c // d) := (a d + b c) // (b d)
    \end{equation}
\end{definition}

\begin{definition}[multiplication]
    \begin{equation}
        (a // b) \times (c // d) := (a c) // (b d)
    \end{equation}
\end{definition}

\begin{definition}[negation]
    \begin{equation}
        -(a // b) := (-a) // b
    \end{equation}
\end{definition}

\begin{lemma}[substitution axiom verified]
    Let $a, b, c, d, a', b'$ be natural numbers.
    If $a // b = a' // b'$, then $a // b + c // d = a' // b' + c // d$, and
    $(a // b) \times (c // d) = (a' // b') \times (c // d)$, and
    $-(a // b) = -(a' // b')$.
\end{lemma}

\begin{definition}[reciprocal]
    If $x = a // b$ is a non-zero rational (so that $a, b \neq 0$), then
    \begin{equation}
        x^{-1} := b // a
    \end{equation}
\end{definition}

\begin{proposition}[field]
    Let $x, y, z$ be rationals.
    \begin{gather}
        x + y = y + z \\
        (x + y) + z = x + (y + z) \\
        x + 0 = 0 + x = x \\
        x + (-x) = (-x) + x = 0 \\
        x y = y x \\
        (x y) z = x (y z) \\
        x 1 = 1 x = x \\
        x (y + z) = x y + x z \\
        (y + z) x = y x + z x \\
        x x^{-1} = x^{-1} x = 1 \text{ for } x \neq 0
    \end{gather}
\end{proposition}

\begin{definition}[quotient]
    Let $x$ and $y$ are two rationals and $y \neq 0$, then
    \begin{equation}
        x / y := x \times y^{-1}
    \end{equation}
\end{definition}

\begin{definition}[substraction]
    \begin{equation}
        x - y := x + (-y)
    \end{equation}
\end{definition}

\begin{definition}[positive and negative]
    A rational number $x$ is positive if and only if $x = a // b$ for some positive integers $a$ and $b$.
    It is said to be negative if and only if $x = -y$ for some positive rational $y$.
\end{definition}

\begin{lemma}[trichotomy of rationals]
    Let $x$ be a rational number. Then exactly one of the following three statements is true:
    (a) $x = 0$;
    (b) $x$ is a positive rational number;
    (c) $x$ is a negative rational number.
\end{lemma}

\begin{definition}[order]
    Let $x$ and $y$ be rational numbers.
    $x > y$ if and only if $x - y$ is a positive rational number, and
    $x < y$ if and only if $x - y$ is a negative rational number.
    $x \geq y$ if and only if either $x > y$ or $x = y$.
\end{definition}

\begin{proposition}[ordered field]
    Let $x, y, z$ be rational numbers, then
    \begin{itemize}
        \item (order trichotomy)
              Exactly one of the three statements $x = y$, $x < y$, $x > y$ is true.
        \item (antisymmetric)
              $x < y$ if and only if $y > x$.
        \item (transitive)
              If $x < y$ and $y < z$, then $x < z$.
        \item (addition preserves order)
              If $x < y$, then $x + z < y + z$.
        \item (positive multiplication preserves order)
              If $x < y$ and $z$ is positive, then $x z < y z$.
    \end{itemize}
\end{proposition}

\subsection{Absolute Value}
\begin{definition}[absolute value]
    If $x$ is a rational number, then the absolute value $|x|$ of $x$ is defined as
    $|x| := x$ if $x$ is positive;
    $|x| := -x$ if $x$ is negative;
    $|x| := 0$ if $x = 0$.
\end{definition}

\begin{definition}[distance]
    Let $x$ and $y$ be rational numbers.
    \begin{equation}
        d(x, y) := |x - y|
    \end{equation}
\end{definition}

\begin{proposition}[properties of absolute value]
    Let $x, y, z$ be rational numbers.
    \begin{itemize}
        \item (non-degeneracy) $|x| = 0$ if and only if $x = 0$.
        \item (triangle inequality) $|x + y| \leq |x| + |y|$.
        \item $-y \leq x \leq y$ if and only if $y \geq |x|$.
              In particular, $-|x| \leq x \leq |x|$.
        \item (multiplicativity) $|x y| = |x| |y|$.
              In particular, $|-x| = |x|$.
    \end{itemize}
\end{proposition}

\begin{proposition}[properties of distance]~
    \begin{itemize}
        \item (non-degeneracy) $d(x, y) \geq 0$.
              $d(x, y) = 0$ if and only if $x = y$.
        \item (symmetry) $d(x, y) = d(y, x)$.
        \item (triangle inequality) $d(x, z) \leq d(x, y) + d(y, z)$.
    \end{itemize}
\end{proposition}

\begin{proof}~
    \begin{itemize}
        \item $d(x, z) = |x - z| = |(x - y) + (y - z)| \leq |x - y| + |y - z| = d(x, y) + d(y, z)$.

              $d(x, z) = d(x - y, z - y) = |(x - y) - (z - y)| \leq |x - y| + |z - y| = d(x, y) + d(z, y)$.
    \end{itemize}
\end{proof}

\begin{definition}[$\varepsilon$-closeness]
    Let $\varepsilon > 0, x, y$ be rational numbers.
    $y$ is $\varepsilon$-close to $x$ if and only if $d(y, x) \leq \varepsilon$.
\end{definition}

\begin{proposition}[properties of $\varepsilon$-close]
    Let $x, y, z, w$ be rational numbers.
    \begin{itemize}
        \item If $x = y$, then $x$ is $\varepsilon$-close to $y$ for every $\varepsilon > 0$.
              Conversely, if $x$ is $\varepsilon$-close to $y$ for every $\varepsilon > 0$,
              then $x = y$.
        \item Let $\varepsilon > 0$. If $x$ is $\varepsilon$-close to $y$,
              then $y$ is $\varepsilon$-close to x.
        \item Let $\varepsilon, \delta > 0$. If $x$ is $\varepsilon$-close to $y$,
              and $y$ is $\delta$-close to $z$,
              then $x$ and $z$ are $(\varepsilon + \delta)$-close.
              \begin{proof}
                  $d(x, z) \leq d(x, y) + d(y, z) \leq \varepsilon + \delta$.
              \end{proof}
        \item Let $\varepsilon, \delta > 0$. If $x$ and $y$ are $\varepsilon$-close,
              and $z$ and $w$ are $\delta$-close,
              then $x + z$ and $y + w$ are $(\varepsilon + \delta)$-close,
              and $x - z$ and $y - w$ are also $(\varepsilon + \delta)$-close.
              \begin{proof}
                  $d(x + z, y + w) \leq d(x + z, y + z) + d(y + z, y + w) = d(x, y) + d(z, w) \leq \varepsilon + \delta$.
              \end{proof}
        \item Let $\varepsilon > 0$. If $x$ and $y$ are $\varepsilon$-close,
              then they are also $\varepsilon'$-close for every $\varepsilon' > \varepsilon$.
              \begin{proof}
                  $d(x, y) \leq \varepsilon < \varepsilon'$.
              \end{proof}
        \item Let $\varepsilon > 0$. If $y$ and $z$ are both $\varepsilon$-close to $x$,
              and $w$ is between $y$ and $z$ (i.e. $y \leq w \leq z$ or $z \leq w \leq y$),
              then $w$ is also $\varepsilon$-close to $x$.
              \begin{lemma}
                  Let $x, y, z$ be rational numbers, such that $x \leq y \leq z$.
                  Then either $|y| \leq |x|$ or $|y| \leq |z|$ (or both).
              \end{lemma}
              \begin{proof}~
                  \begin{itemize}
                      \item If $y = 0$, then $|y| = 0 \leq |x|$ and $|y| = 0 \leq |z|$.
                      \item If $y > 0$, then $0 < y \leq z$, then $|y| \leq |z|$.
                      \item If $y < 0$, then $-y \leq -x < 0$, then $|y| \leq |x|$.
                  \end{itemize}
              \end{proof}
              \begin{proof}
                  Suppose $y \leq w \leq z$, then
                  \begin{equation*}
                      y - x \leq w - x \leq z - x
                  \end{equation*}
                  Then either
                  \begin{equation*}
                      |w-x|\leq |y-x|
                  \end{equation*}
                  or
                  \begin{equation*}
                      |w-x|\leq |z-x|
                  \end{equation*}
                  Then either
                  \begin{equation*}
                      d(w,x) \leq d(y,x) \leq \varepsilon
                  \end{equation*}
                  or
                  \begin{equation*}
                      d(w,x) \leq d(z,x) \leq \varepsilon
                  \end{equation*}
              \end{proof}
        \item Let $\varepsilon > 0$. If $x$ and $y$ are $\varepsilon$-close, and $z$ is non-zero,
              then $x z$ and $y z$ are $\varepsilon|z|$-close.
              \begin{proof}
                  $d(x z, y z) = |x z - y z| = |z| |x - y| \leq |z| \varepsilon$.
              \end{proof}
        \item Let $\varepsilon, \delta > 0$. If $x$ and $y$ are $\varepsilon$-close,
              and $z$ and $w$ are $\delta$-close,
              then $x z$ and $y w$ are $(\varepsilon |z| + \delta |x| + \varepsilon \delta)$-close.
              \begin{proof}
                  \begin{equation*}
                      d(x z, y w) \leq d(x z, y z) + d(y z, y w) \leq |z| d(x, y) + |y| d(z, w)
                  \end{equation*}
                  Since
                  \begin{equation*}
                      d(y, 0) \leq d(y, x) + d(x, 0)
                  \end{equation*}
                  then
                  \begin{equation*}
                      d(x z, y w) \leq |z| d(x, y) + (d(y, x) + |x|) d(z, w)
                      \leq |z| \varepsilon + \varepsilon \delta + |x| \delta
                  \end{equation*}
              \end{proof}
    \end{itemize}
\end{proposition}

\subsection{Exponentiation}
\begin{definition}[exponentiation]
    Let $x$ be a rational number.
    $x^{0} := 1$. In particular, $0^{0} := 1$.
    $x^{n + 1} := x^{n} \times x$.
\end{definition}

\begin{proposition}[properties of exponentiation]
    Let $x$ and $y$ be rational numbers.
    \begin{itemize}
        \item $x^{n} x^{m} = x^{n + m}$, $(x^{n})^{m} = x^{nm}$, and $(x y)^{n} = x^{n} y^{n}$.
        \item Suppose $n > 0$. Then $x^{n} = 0$ if and only if $x = 0$.
        \item If $x \geq y \geq 0$, then $x^{n} \geq y^{n} \geq 0$.
              If $x > y \geq 0$ and $n > 0$, then $x^{n} > y^{n} \geq 0$.
        \item $|x^{n}| = |x|^{n}$.
    \end{itemize}
\end{proposition}

\begin{definition}[negative exponentiation]
    Let $x$ be a non-zero rational number.
    Then for any negative integer $-n$, $x^{-n} := 1 / x^{n}$.
\end{definition}

\begin{proposition}[properties of exponentiation]
    Let $x$ and $y$ be non-zero rational numbers, and let $n$ and $m$ be integers.
    \begin{itemize}
        \item $x^{n} x^{m} = x^{n + m}$, $(x^{n})^{m} = x^{nm}$, and $(x y)^{n} = x^{n} y^{n}$.
        \item If $x \geq y > 0$, then $x^{n} \geq y^{n} > 0$ if $n$ is positive,
              and $0 < x^{n} < y^{n}$ if $n$ is negative.
        \item If $x, y > 0, n \neq 0, x^{n} = y^{n}$, then $x = y$
        \item $|x^{n}| = |x|^{n}$.
    \end{itemize}
\end{proposition}

\subsection{Gaps in Rational Numbers}
\begin{proposition}[integer part]
    Let $x$ be a rational number.
    Then there exists a unique integer $n$ such that $n \leq x < n + 1$.
    In particular there exists a natural number $N$ such that $N > x$.
\end{proposition}

\begin{proof}
    Existence:
    \begin{itemize}
        \item If $x = 0$, then $0 \leq x < 1$.
        \item If $x > 0$, then suppose $x = a / b = (m b + r) / b = m + r / b$,
              where $0 \leq r < b$.
              Then $m \leq x < m + 1$.
        \item If $x < 0$, then suppose $x = -m - r / b$.
              Then $-m - 1 \leq x < -m$.
    \end{itemize}
    Uniqueness:

    Suppose
    \begin{equation*}
        n_{1} \leq x < n_{1} + 1
    \end{equation*}
    and
    \begin{equation*}
        n_{2} \leq x < n_{2} + 1
    \end{equation*}
    Suppose $n_{1} < n_{2}$, then
    \begin{equation*}
        n_{1} + 1 \leq n_{2} \leq x < n_{1} + 1
    \end{equation*}
    Contradict.
\end{proof}

\begin{proposition}[intersperse]
    If $x$ and $y$ are two rational numbers such that $x < y$.
    Then there exists a third rational $z$ such that $x < z < y$.
\end{proposition}

\begin{proof}
    $z = (x + y) / 2$.
\end{proof}

\begin{definition}[infinite descent]
    A sequence $a_{0}, a_{1}, a_{2}, \dots$ of numbers is said to be infinite descent if
    $a_{n} > a_{n + 1}$ for all natural numbers $n$.
\end{definition}

\begin{lemma}[principle of infinite descent]
    It is not possible to have a sequence of natural numbers which is infinite descent.
\end{lemma}

\begin{proof}
    Suppose for sake of contradiction that
    $a_{0} > a_{1} > a_{2} > \dots$ is a sequence of natural numbers which is infinite descent,
    then $a_{n} \geq k$ for all $k \in \mathbb{N}$ and all $n \in \mathbb{N}$.

    When $k = 0$, $a_{n} \geq 0$ for all $n \in \mathbb{N}$.

    Suppose $a_{n} \geq k$ for all $n \in \mathbb{N}$.

    Then
    \begin{gather*}
        a_{n} > a_{n + 1} \geq k \\
        a_{n} \geq k + 1
    \end{gather*}

    This closes the induction.

    When $k = a_{n} + 1$,
    \begin{equation*}
        a_{n} \geq a_{n} + 1
    \end{equation*}
    Contradiction.
\end{proof}

\begin{proposition}[$\sqrt{2}$]
    There does not exist any rational number $x$ for which $x^{2} = 2$.
\end{proposition}

\begin{proof}
    Suppose $x = p / q$, where $p, q$ are natural numbers.

    Then
    \begin{gather*}
        x^{2} = p^{2} / q^{2} = 2 \\
        p^{2} = 2 q^{2}
    \end{gather*}
    $p > q$ and $p$ is even.

    Let $p = 2 k$, then
    \begin{gather*}
        4 k^{2} = 2 q^{2} \\
        q^{2} = 2 k^{2}
    \end{gather*}
    $q > k$ and $q$ is even.

    Repeat this process
    \begin{equation*}
        p > q > k > ...
    \end{equation*}
    Contradict with principle of infinite descent.
\end{proof}

\begin{proposition}
    For every rational number $\varepsilon > 0$,
    there exists a non-negative rational number $x$ such that $x^{2} < 2 < (x + \varepsilon)^{2}$.
\end{proposition}

\begin{proof}
    When $\varepsilon \geq 1$,
    \begin{equation*}
        1^{2} < 2 < (1 + \varepsilon)^2.
    \end{equation*}

    When $0 < \varepsilon < 1$, suppose for sake of contradiction that there is no $x$ for which
    $x^{2} < 2 < (x + \varepsilon)^{2}$.

    Then for every $x^{2} < 2$,
    \begin{equation*}
        (x + \varepsilon)^{2} < 2
    \end{equation*}

    Let $x = 0$, then
    \begin{equation*}
        (0 + \varepsilon)^{2} = \varepsilon^{2} < 2
    \end{equation*}

    Suppose $(n \varepsilon)^{2} < 2$, then
    \begin{equation*}
        (n \varepsilon + \varepsilon)^{2} = ((n + 1) \varepsilon) < 2
    \end{equation*}
    This closes induction.

    Let $n > 2 / \varepsilon$, then
    \begin{equation*}
        (n \varepsilon)^{2} > 4
    \end{equation*}
    Contradiction.
\end{proof}

\subsection{Cauchy Sequences}
\begin{definition}[sequence]
    A sequence $(a_{n})_{n = m}^{\infty}$ of rational numbers
    is a collection of rationals $a_{m}, a_{m + 1}, a_{m + 2}, \dots$.
\end{definition}

\begin{definition}[$\varepsilon$-steadiness]
    Let $\varepsilon > 0$.
    The sequence $a_{0}, a_{1}, a_{2}, \dots$ is $\varepsilon$-steady
    if and only if $d(a_{j}, a_{k}) \leq \varepsilon$ for all $j, k$.
\end{definition}

\begin{definition}[eventual $\varepsilon$-steadiness]
    Let $\varepsilon > 0$.
    The sequence $a_{0}, a_{1}, a_{2}, \dots$ is eventual $\varepsilon$-steady
    if and only if there exists a natural number $N \geq 0$ such that
    $d(a_{j}, a_{k}) \leq \varepsilon$ for all $j, k \geq N$.
\end{definition}

\begin{definition}[Cauchy sequence]
    A sequence $(a_{n})_{n = 0}^{\infty}$ is a Cauchy sequence if and only if for every rational $\varepsilon > 0$,
    the sequence $(a_{n})_{n = 0}^{\infty}$ is eventually $\varepsilon$-steady.
\end{definition}

\begin{definition}
    The sequence $(1 / n)_{n = 1}^{\infty}$ is a Cauchy sequence.
\end{definition}

\begin{proof}
    For all $j, k \geq N$,
    \begin{gather*}
        0 < 1 / j, 1 / k \leq 1 / N \\
        |1 / j  - 1 / k| \leq 1 / N
    \end{gather*}
    Choose an $N$ such that $1 / N \leq \varepsilon$ or $N \geq 1 / \varepsilon$.
\end{proof}

\begin{definition}[bounded sequence]
    Let $M \geq 0$ be rational.
    A finite sequence $a_{1}, a_{2}, \dots, a_{n}$ is bounded by $M$ if and only if
    $|a_{i}| \leq M$ for all $1 \leq i \leq n$.
    A infinite sequence $(a_{n})_{n = 1}^{\infty}$ is bounded by $M$ if and only if
    $|a_{i}| \leq M$ for all $i \geq 1$.
    A sequence is bounded if and only if it is bounded by $M$ for some rational $M \geq 0$.
\end{definition}

\begin{lemma}[finite sequences are bounded]
    Every finite sequence $a_{1}, a_{2}, \dots, a_{n}$ is bounded.
\end{lemma}

\begin{proof}
    When $n = 1$,
    \begin{equation*}
        |a_{1}| \leq |a_{1}|
    \end{equation*}

    Suppose $a_{1}, a_{2}, \dots, a_{n}$ is bounded by $M$, then
    \begin{equation*}
        |a_{i}| \leq M \text{ for all } 1 \leq i \leq n
    \end{equation*}

    Then
    \begin{equation*}
        |a_{i}| \leq M + |a_{n + 1}| \text{ for all } 1 \leq i \leq n + 1
    \end{equation*}
    This closes induction.
\end{proof}

\begin{lemma}[Cauchy sequences are bounded]
    Every Cauchy sequence $(a_{n})_{n = 1}^{\infty}$ is bounded.
\end{lemma}

\begin{proof}
    For every $\varepsilon > 0$, there exists a natural number $N \geq 1$ such that
    $|a_{j} - a_{k}| \leq \varepsilon$ for all $j, k \geq N$.

    Then
    \begin{equation*}
        d(a_{j}, 0) \leq d(a_{j}, a_{k}) + d(a_{k}, 0) = |a_{j} - a_{k}| + |a_{k}| \leq \varepsilon + |a_{k}|
        \text{ for all } j \geq N
    \end{equation*}

    For $n < N$, sequence $a_{1}, a_{2}, \dots, a_{n}$ is bounded.
\end{proof}

\subsection{Equivalent Sequences}
\begin{definition}[$\varepsilon$-close sequences]
    Let $(a_{n})_{n = 0}^{\infty}$ and $(b_{n})_{n = 0}^{\infty}$ be two sequences,
    and let $\varepsilon > 0$.
    The sequence $a_{0}, a_{1}, a_{2}, \dots$ is
    $\varepsilon$-close to the sequence $b_{0}, b_{1}, b_{2}, \dots$
    if and only if $d(a_{n}, b_{n}) \leq \varepsilon$ for all $n = 0, 1, 2, \dots$.
\end{definition}

\begin{definition}[eventually $\varepsilon$-close sequences]
    Let $(a_{n})_{n = 0}^{\infty}$ and $(b_{n})_{n = 0}^{\infty}$ be two sequences,
    and let $\varepsilon > 0$.
    The sequence $a_{0}, a_{1}, a_{2}, \dots$ is
    eventually $\varepsilon$-close to the sequence $b_{0}, b_{1}, b_{2}, \dots$
    if and only if there exists an $N \geq 0$ such that
    $d(a_{n}, b_{n}) \leq \varepsilon$ for all $n \geq N$.
\end{definition}

\begin{definition}[equivalent sequences]
    Two sequences $(a_{n})_{n = 0}^{\infty}$ and $(b_{n})_{n = 0}^{\infty}$ are equivalent
    if and only if for every rational $\varepsilon > 0$,
    the sequences $(a_{n})_{n = 0}^{\infty}$ and $(b_{n})_{n = 0}^{\infty}$ are eventually $\varepsilon$-close.
\end{definition}

\begin{lemma}
    \begin{equation*}
        10^{-m} < 10^{-n} \text{ whenever } m > n
    \end{equation*}
\end{lemma}

\begin{proof}
    Since $m > n$, then
    \begin{equation*}
        m = n + k
    \end{equation*}

    When $k = 1$,
    \begin{equation*}
        10^{-(n + 1)} < 10^{-n}
    \end{equation*}

    Suppose
    \begin{equation}
        10^{-(n + k)} < 10^{-n}
    \end{equation}

    Then
    \begin{equation*}
        10^{-(n + k + 1)} < 10^{-n}
    \end{equation*}
    This closes induction.
\end{proof}

\begin{lemma}
    \begin{equation*}
        10^{-n} \leq 1 / n \text{ for } n > 0
    \end{equation*}
\end{lemma}

\begin{proof}
    When $n = 1$,
    \begin{equation*}
        10^{-1} \leq 1
    \end{equation*}

    Suppose
    \begin{equation*}
        10^{-n} \leq 1 / n
    \end{equation*}

    Then
    \begin{equation*}
        10^{-(n + 1)} \leq 1 / n < 1 / (n + 1)
    \end{equation*}
    This closes induction.
\end{proof}

\begin{proposition}
    Let $(a_{n})_{n = 0}^{\infty}$ and $(b_{n})_{n = 0}^{\infty}$ be the sequences
    $a_{n} = 1 + 10^{-n}$ and $b_{n} = 1 - 10^{-n}$. Then the sequences $a_{n}, b_{n}$ are equivalent.
\end{proposition}

\begin{proof}
    For all $n \geq N$,
    \begin{gather*}
        d(a_{n}, b_{n}) = |a_{n} - b_{n}| = 2 \times 10^{-n} \leq 2 \times 10^{-N}
    \end{gather*}
    Choose $N$ such that
    \begin{equation*}
        2 \times 10^{-N} \leq 2 / N \leq \varepsilon
    \end{equation*}
\end{proof}

\section{Tools}
\begin{definition}[implication]
    \label{def:implication}
    ``If P, then Q'' = ``If not Q, then not P''.
\end{definition}

\begin{lemma}[de mongan laws]~
    \begin{itemize}
        \item not ``A and B'' = ``not A or not B''
        \item not ``A or B'' = ``not A and not B''
    \end{itemize}
\end{lemma}

\begin{axiom}[subsititution axiom]
    Given any two objects $x$ and $y$, if $x = y$, then $f(x) = f(y)$ for all functions or operations.
    For any property $P(x)$ pertaining to $x$, if $x=y$ , then $P(x)$ and $P(y)$ are equivalent statements.
\end{axiom}

\begin{definition}[conjunction and disjunction]~
    \begin{itemize}
        \item The statement ``X and Y'' is true if $X$ and $Y$ are both true.
        \item The statement ``X or Y'' is true if either $X$ or $Y$ is true, or both.
    \end{itemize}
\end{definition}
