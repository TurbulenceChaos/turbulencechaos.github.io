\section{Natural Number}
\begin{definition}[natural number]
    A natural number is any element of the set
    \begin{equation}
        \mathbb{N} := {0, 1, 2, 3, \dots}
    \end{equation}
\end{definition}

\begin{axiom}[0]
    0 is a natural number.
\end{axiom}

\begin{axiom}[successor]
    If $n$ is a natural number, then $n{++}$ is also a natural number.
\end{axiom}

\begin{definition}[notation]
    $1 := 0{++}$, $2 := 1{++}$, $3 := 2{++}$, $\dots$
\end{definition}

\begin{axiom}[$n{++} \neq 0$]
    0 is not the successor of any natural number $n$.
\end{axiom}

\begin{axiom}[$n{++} \neq m{++}$]
    Different natural numbers must have different successors.

    If $n \neq m$, then $n{++} \neq m{++}$.
    Equivalently, if $n{++} = m{++}$, then $n = m$.
\end{axiom}

\begin{axiom}[principle of mathematical induction]
    Let $P(n)$ be any property pertaining to a natural number $n$.
    Suppose that $P(0)$ is true, and suppose that whenever $P(n)$ is true,
    $P(n{++})$ is also true. Then $P(n)$ is true for every natural number $n$.
\end{axiom}

\begin{lemma}[recursive definition of sequence]
    $a_{0} = c$ and $a_{n{++} = f_{n}(a_{n})}$ for each natural number $n$.
\end{lemma}

\subsection{Addition}
\begin{definition}[addition]
    Define $0+m:=m$. Suppose inductively that we have defined $n+m$.
    Then $(n{++})+m:=(n+m){++}$.
\end{definition}

$a_{n} = n+m$ and $f_{n}(a_{n}) = a_{n}{++}$

\begin{lemma}
    $n+0=n$
\end{lemma}

\begin{lemma}
    $n+(m{++})=(n+m){++}$
\end{lemma}

\begin{proposition}[addition is commutative]
    \begin{equation}
        n+m = m+n
    \end{equation}
\end{proposition}

\begin{proposition}[addition is associative]
    \begin{equation}
        (a+b)+c = a+(b+c)
    \end{equation}
\end{proposition}

\begin{proposition}[cancellation law]
    If $a+b = a+c$, then $b=c$.
\end{proposition}

\begin{definition}
    A natural number $n$ is positive if and only if $n \neq 0$.
\end{definition}

\begin{lemma}
    If $a$ is a positive natural number and $b$ is a natural number,
    then $a+b$ is positive.
\end{lemma}

\begin{lemma}
    If $a+b = 0$, then $a=0$ and $b=0$.
\end{lemma}

\begin{lemma}
    If $a$ is a positive number, then there exists exactly one natural number $b$
    such that $b{++} = a$.
\end{lemma}

\begin{definition}[order]~
    \begin{itemize}
        \item $n \geq m$ or $m \leq n$ if and only if $n=m+a$ for some natural number $a$
        \item $n>m$ or $m<n$ if and only if $n \geq m$ and $n \neq m$
    \end{itemize}
\end{definition}

\begin{proposition}[properties of order]~
    \begin{itemize}
        \item reflexive: $a \geq a$
        \item transitive: If $a \geq b$ and $b \geq c$, then $a \geq c$
        \item antisymmetric: If $a \geq b$ and $b \geq a$, then $a=b$
        \item addition preserve order: $a \geq b$ if and only if $a+c \geq b+c$
        \item $a<b$ if and only if $a{++} \leq b$
        \item $a<b$ if and only if $b=a+d$ for some positive number $d$
    \end{itemize}
\end{proposition}

\begin{proposition}[trichotomy of order]
    Exactly one of following statement is true:
    $a<b$, $a=b$, or $a>b$.
\end{proposition}

\begin{proposition}[strong principle of induction]
    Let $m_{0}$ be a natural number.
    Suppose that for each $m \geq m_{0}$, if $P(m')$ is true for all $m_{0} \leq m' < m$,
    then $P(m)$ is also true. Then $P(m)$ is true for all $m \geq m_{0}$.
\end{proposition}

\begin{example}[Fibonacci sequence]
    \begin{gather}
        F_{0} = 0 \\
        F_{1} = 1 \\
        F_{n} = F_{n-1} + F_{n-2}
    \end{gather}
\end{example}

\subsection{Multiplication}
\begin{definition}[multiplication]
    Define $0 \times m := 0$.
    Suppose inductively that we have defined $n \times m$.
    Then $(n{++}) \times m := (n \times m) + m$.
\end{definition}

\begin{lemma}
    $n \times 0 = 0 \times n$
\end{lemma}

\begin{lemma}
    $n \times (m{++}) = (n \times m) + n$
\end{lemma}

\begin{proposition}[multiplication is commutative]
    \begin{equation}
        n \times m = m \times n
    \end{equation}
\end{proposition}

\begin{lemma}[zero divisor]
    $n \times m = 0$ if and only if at least one of $n$ and $m$ is equal to 0.
    If $n$ and $m$ are both positive, then $nm$ is also positive.
\end{lemma}

\begin{proposition}[distributive law]
    \begin{equation}
        a(b+c)=(b+c)a=bc+ca
    \end{equation}
\end{proposition}

\begin{proposition}[multiplication is associative]
    \begin{equation}
        (ab)c=a(bc)
    \end{equation}
\end{proposition}

\begin{lemma}[multiplication preserve order]
    If $a<b$ and $c$ is positive, then $ac<nc$.
\end{lemma}

\begin{proposition}[cancellation law]
    If $ac=bc$ and $c$ is positive, then $a=b$.
\end{proposition}

\begin{proposition}[Euclid's division lemma]
    Let $n$ be a natural number and $q$ be a positive natural number.
    Then there exist natural numbers $m$ and $r$, such that $0 \leq r < q$ and $n=mq+r$.
\end{proposition}

\begin{proof}
    $0 = 0 \times q + 0$ and $0 \leq r < q$.
    Suppose that $n=mq+r$ and $0 \leq r < q$.
    Then $n{++}=mq+(r{++})$ and $0 \leq r{++} \leq q$.
    If $r{++}=q$, $n{++}=mq+q=(m+1)q+0$.
    If $r{++}<q$, $n{++}=mq+r{++}$.
\end{proof}

\begin{definition}[Exponentiation]
    Define $m^{0}:=1$, in particular $0^{0}:=1$.
    Suppose recursively that $m^{n}$ has been defined for some natural number $n$,
    then define $m^{n{++}}:=m^{n} \times m$.
\end{definition}

\section{Set Theory}
\begin{definition}[set]
    A set $A$ is any unordered collection of objects.
    If $x$ is an element of A, then $x \in A$, otherwise $x \notin A$.
\end{definition}

\begin{axiom}[sets are objects]
    If $A$ is a set, then $A$ is also an object.
\end{axiom}

\begin{axiom}[equality of sets]
    $A=B$ if and only if every element of $A$ is an element of $B$ and vice versa.
\end{axiom}

\begin{axiom}[empty set]
    There exists an empty set $\emptyset$ which contains no elements.
\end{axiom}

\begin{lemma}[single choice]
    If $A$ is a non-empty set, then there exists an object $x$ such that $x \in A$.
\end{lemma}

\begin{axiom}[singleton and pair sets]
    If $a$ is an object, then there exists a singleton set $\{a\}$ whose only element is $a$.
    If $a$ and $b$ are objects, then there exists a pair set $\{a,b\}$ whose only elements are $a$ and $b$.
\end{axiom}

\begin{axiom}[pairwise union]
    If $A$ and $B$ are two sets, then there exists a union of $A$ and $B$ set $A \cup B$,
    which consists of all the elements which belong to $A$ or $B$ or both.
\end{axiom}

\begin{definition}[subsets]
    $A$ is a subset of $B$, denoted $A \subseteq B$,
    if and only if every element of $A$ is also an element of $B$.

    $A$ is a proper subset of $B$, denoted $A \subsetneq B$,
    if $A \subseteq B$ and $A \neq B$.
\end{definition}

\begin{axiom}[axiom of specification]
    Let $A$ be a set and $P(x)$ be a property pertaining to $x$ for each $x \in A$.
    Then there exists a set $\{x \in A : P(x) \text{ is true}\}$
    whose elements are precisely the elements $x$ in $A$ for which $P(x)$ is true.
\end{axiom}

\begin{definition}[intersections]
    The intersection of two sets is defined to be the set
    \begin{equation}
        A \cap B := \{x \in A : x \in B\}
    \end{equation}
\end{definition}

\begin{definition}[difference sets]
    Define $A-B$ or $A \backslash B$ to be set $A$ with elements of $B$ removed
    \begin{equation}
        A \backslash B := \{x \in A : x \notin B\}
    \end{equation}
\end{definition}

\begin{axiom}[replacement]
    If $P(x,y)$ is a property pertaining to $x$ and $y$,
    such that for each $x \in A$ there is at most one $y$ for which $P(x,y)$ is true.
    Then there exists a set $\{y : P(x,y) \text{ is true for some } x \in A\}$.
\end{axiom}

\begin{axiom}[infinity]
    There exists a set $\mathbb{N}$, whose elements are called natural numbers,
    as well as an object 0 in $\mathbb{N}$,
    and an object $n{++}$ assigned to every $n \in \mathbb{N}$,
    such that the Peano axioms hold.
\end{axiom}
