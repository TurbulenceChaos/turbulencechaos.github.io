\section{Natural Number}
\begin{definition}[natural number]
    A natural number is any element of the set
    \begin{equation}
        \mathbb{N} := {0, 1, 2, 3, \dots}
    \end{equation}
\end{definition}

\begin{axiom}[0]
    \label{axiom:peano-axiom-1}
    0 is a natural number.
\end{axiom}

\begin{axiom}[successor]
    \label{axiom:peano-axiom-2}
    If $n$ is a natural number, then $n{++}$ is also a natural number.
\end{axiom}

\begin{definition}
    \label{def:nat}
    $1 := 0{++}$, $2 := 1{++}$, $3 := 2{++}$, $\dots$
\end{definition}

\begin{axiom}[$n{++} \neq 0$]
    \label{axiom:peano-axiom-3}
    0 is not the successor of any natural number $n$.
\end{axiom}

\begin{axiom}[$n{++} \neq m{++}$]
    \label{axiom:peano-axiom-4}
    Different natural numbers must have different successors.

    If $n \neq m$, then $n{++} \neq m{++}$.
    Equivalently \ref{def:implication}, if $n{++} = m{++}$, then $n = m$.
\end{axiom}

\begin{axiom}[principle of mathematical induction]
    \label{axiom:peano-axiom-5}
    Let $P(n)$ be any property pertaining to a natural number $n$.
    Suppose that $P(0)$ is true, and suppose that whenever $P(n)$ is true,
    $P(n{++})$ is also true. Then $P(n)$ is true for every natural number $n$.
\end{axiom}

\begin{lemma}[recursive definition of sequence]
    $a_{0} = c$ and $a_{n{++} = f_{n}(a_{n})}$ for each natural number $n$.
\end{lemma}

\subsection{Addition}
\begin{definition}[addition]
    Define $0+m:=m$. Suppose inductively that we have defined $n+m$.
    Then $(n{++})+m:=(n+m){++}$.
\end{definition}

$a_{n} = n+m$ and $f_{n}(a_{n}) = a_{n}{++}$

\begin{lemma}
    $n+0=n$
\end{lemma}

\begin{lemma}
    $n+(m{++})=(n+m){++}$
\end{lemma}

\begin{proposition}[addition is commutative]
    \begin{equation}
        n+m = m+n
    \end{equation}
\end{proposition}

\begin{proposition}[addition is associative]
    \begin{equation}
        (a+b)+c = a+(b+c)
    \end{equation}
\end{proposition}

\begin{proposition}[cancellation law]
    If $a+b = a+c$, then $b=c$.
\end{proposition}

\begin{definition}
    A natural number $n$ is positive if and only if $n \neq 0$.
\end{definition}

\begin{lemma}
    If $a$ is a positive natural number and $b$ is a natural number,
    then $a+b$ is positive.
\end{lemma}

\begin{lemma}
    If $a+b = 0$, then $a=0$ and $b=0$.
\end{lemma}

\begin{lemma}
    If $a$ is a positive number, then there exists exactly one natural number $b$
    such that $b{++} = a$.
\end{lemma}

\begin{definition}[order]~
    \begin{itemize}
        \item $n \geq m$ or $m \leq n$ if and only if $n=m+a$ for some natural number $a$
        \item $n>m$ or $m<n$ if and only if $n \geq m$ and $n \neq m$
    \end{itemize}
\end{definition}

\begin{proposition}[properties of order]~
    \begin{itemize}
        \item reflexive: $a \geq a$
        \item transitive: If $a \geq b$ and $b \geq c$, then $a \geq c$
        \item antisymmetric: If $a \geq b$ and $b \geq a$, then $a=b$
        \item addition preserve order: $a \geq b$ if and only if $a+c \geq b+c$
        \item $a<b$ if and only if $a{++} \leq b$
        \item $a<b$ if and only if $b=a+d$ for some positive number $d$
    \end{itemize}
\end{proposition}

\begin{proposition}[trichotomy of order]
    Exactly one of following statement is true:
    $a<b$, $a=b$, or $a>b$.
\end{proposition}

\begin{proposition}[strong principle of induction]
    Let $m_{0}$ be a natural number.
    Suppose that for each $m \geq m_{0}$, if $P(m')$ is true for all $m_{0} \leq m' < m$,
    then $P(m)$ is also true. Then $P(m)$ is true for all $m \geq m_{0}$.
\end{proposition}

\begin{example}[Fibonacci sequence]
    \begin{gather}
        F_{0} = 0 \\
        F_{1} = 1 \\
        F_{n} = F_{n-1} + F_{n-2}
    \end{gather}
\end{example}

\subsection{Multiplication}
\begin{definition}[multiplication]
    Define $0 \times m := 0$.
    Suppose inductively that we have defined $n \times m$.
    Then $(n{++}) \times m := (n \times m) + m$.
\end{definition}

\begin{lemma}
    $n \times 0 = 0 \times n$
\end{lemma}

\begin{lemma}
    $n \times (m{++}) = (n \times m) + n$
\end{lemma}

\begin{proposition}[multiplication is commutative]
    \begin{equation}
        n \times m = m \times n
    \end{equation}
\end{proposition}

\begin{lemma}[zero divisor]
    $n \times m = 0$ if and only if at least one of $n$ and $m$ is equal to 0.
    If $n$ and $m$ are both positive, then $nm$ is also positive.
\end{lemma}

\begin{proposition}[distributive law]
    \begin{equation}
        a(b+c)=(b+c)a=bc+ca
    \end{equation}
\end{proposition}

\begin{proposition}[multiplication is associative]
    \begin{equation}
        (ab)c=a(bc)
    \end{equation}
\end{proposition}

\begin{lemma}[multiplication preserve order]
    If $a<b$ and $c$ is positive, then $ac<nc$.
\end{lemma}

\begin{proposition}[cancellation law]
    If $ac=bc$ and $c$ is positive, then $a=b$.
\end{proposition}

\begin{proposition}[Euclid's division lemma]
    Let $n$ be a natural number and $q$ be a positive natural number.
    Then there exist natural numbers $m$ and $r$, such that $0 \leq r < q$ and $n=mq+r$.
\end{proposition}

\begin{proof}
    $0 = 0 \times q + 0$ and $0 \leq r < q$.
    Suppose that $n=mq+r$ and $0 \leq r < q$.
    Then $n{++}=mq+(r{++})$ and $0 \leq r{++} \leq q$.
    If $r{++}=q$, $n{++}=mq+q=(m+1)q+0$.
    If $r{++}<q$, $n{++}=mq+r{++}$.
\end{proof}

\begin{definition}[Exponentiation]
    Define $m^{0}:=1$, i.e. $0^{0}:=1$.
    Suppose recursively that $m^{n}$ has been defined for some natural number $n$,
    then define $m^{n{++}}:=m^{n} \times m$.
\end{definition}

\section{Set Theory}
\begin{definition}
    Set $A$ is any unordered collection of objects.
    $x \in A$ if $x$ lies in the collection, otherwise $x \notin A$.
\end{definition}

\begin{axiom}~
    \label{axiom:sets}
    \begin{enumerate}
        \item \label{item:sets-1} If $A$ is a set, then $A$ is also an object.
        \item \label{item:sets-2} $A=B$ if and only if every element of $A$ belongs also to $B$ and vice versa.
        \item \label{item:sets-3} There exists an empty set $\emptyset$, such that every object $x \notin \emptyset$.
        \item \label{item:sets-4} Let $A$ be a non-empty set. Then there exist an object $x \in A$.
        \item \label{item:sets-5} Singleton set $\{a\}$: $y \in \{a\}$ if and only if $y=a$.
              Pair set $\{a,b\}$: $y \in \{a,b\}$ if and only if $y=a$ or $y=b$.
        \item \label{item:sets-6} $x \in A \cup B \Leftrightarrow x \in A$ or $x \in B$.
        \item \label{item:sets-7} $y \in \{x \in A : P(x) \text{ is true}\} \Leftrightarrow y \in A \text{ and } P(x) \text{ is true}$.
        \item \label{item:sets-8} $z \in \{y : P(x,y) \text{ is true for some } x \in A\} \Leftrightarrow P(x,z) \text{ is true for some } x \in A$.
        \item \label{item:sets-9} There exists a set $\mathbb{N}$, whose elements are called natural numbers, and an object $n{++}$ assigned to every $n \in \mathbb{N}$, such that the Peano axioms \ref{axiom:peano-axiom} hold.
    \end{enumerate}
\end{axiom}

\begin{example}
    $\{a,a\} = \{a\}$
\end{example}

\begin{example}
    $\emptyset$, $\{\emptyset\}$, $\{\{\emptyset\}\}$, $\{\emptyset,\{\emptyset\}\}$ are all distinct.
\end{example}

\begin{proof}
    $y \notin \emptyset$.

    $y=\emptyset \in \{\emptyset\}$.

    $y=\{\emptyset\} \in \{\{\emptyset\}\}$.

    $y=\emptyset$ or $y=\{\emptyset\} \in \{\emptyset,\{\emptyset\}\}$.
\end{proof}

\begin{lemma}
    Let $a$, $b$ are objects, and $A$, $B$, $C$ are sets.
    \begin{enumerate}
        \item $\{a,b\}=\{a\}\cup\{b\}$
        \item $A \cup B = B \cup A$
        \item $(A \cup B) \cup C = A \cup (B \cup C)$
    \end{enumerate}
\end{lemma}

\begin{definition}
    \begin{gather*}
        \{a,b,c\} := \{a\}\cup\{b\}\cup\{c\} \\
        \{a,b,c,d\} := \{a\}\cup\{b\}\cup\{c\}\cup\{d\} \\
        \vdots
    \end{gather*}
\end{definition}

\begin{definition}[Subsets]
    \begin{enumerate}
        \item $A \subseteq B \Leftrightarrow$ for any object $x \in A$, then $x \in B$.
        \item $A \subsetneq B$ if $A \subseteq B$ and $A \neq B$.
    \end{enumerate}
\end{definition}

\begin{lemma}
    Let $A$, $B$, $C$ be sets.
    \begin{enumerate}
        \item If $A \subseteq B$ and $B \subseteq C$, then $A \subseteq C$.
        \item If $A \subseteq B$ and $B \subseteq A$, then $A = C$.
        \item If $A \subsetneq B$ and $B \subsetneq C$, then $A \subsetneq C$.
    \end{enumerate}
\end{lemma}

\begin{definition}
    Let $A$, $B$ be sets.
    \begin{enumerate}
        \item $A \cap B := \{x \in A : x \in B\}$.
        \item $A \backslash B := \{x \in A : x \notin B\}$.
    \end{enumerate}
\end{definition}

\begin{lemma}
    \label{prop:sets-boolean-algebra}
    Let $A$, $B$, $C$ be sets, and $\{A,B,C\} \subseteq X$.
    \begin{enumerate}
        \item \label{item:sets-boolean-algebra-1} $A \cup \emptyset = A$ and $A \cap \emptyset = \emptyset$.
        \item \label{item:sets-boolean-algebra-2} $A \cup X = X$ and $A \cap X = A$.
        \item \label{item:sets-boolean-algebra-3} $A \cup A = A$ and $A \cap A = A$.
        \item \label{item:sets-boolean-algebra-4} $A \cup B = B \cup A$ and $A \cap B = B \cap A$.
        \item \label{item:sets-boolean-algebra-5} $(A \cup B) \cup C = A \cup (B \cup C)$ and $(A \cap B) \cap C = A \cap (B \cap C)$.
        \item \label{item:sets-boolean-algebra-6} $A \cap (B \cup C) = (A \cap B) \cup (A \cap C)$ and $A \cup (B \cap C) = (A \cup B) \cap (A \cup C)$.
        \item \label{item:sets-boolean-algebra-7} $A \cup (X \backslash A) = X$ and $A \cap (X \backslash A) = \emptyset$.
        \item \label{item:sets-boolean-algebra-8} De Morgan laws: $X \backslash (A \cup B) = (x \backslash A) \cap (X \backslash B)$ and $X \backslash (A \cap B) = (X \backslash A) \cup (X \backslash B)$.
    \end{enumerate}
\end{lemma}

\begin{proof}
    Prop. \ref{prop:sets-boolean-algebra} (\ref{item:sets-boolean-algebra-5}).

    $x \in A \cap X \Leftrightarrow x \in A \Rightarrow A \cap X = A$.
\end{proof}

\begin{proof}
    Prop. \ref{prop:sets-boolean-algebra} (\ref{item:sets-boolean-algebra-6}).

    $x \in A \cap (B \cup C) \Rightarrow x \in A \text{ and } (x \in B \text{ or } x \in C) \Rightarrow x \in A \cap B \text{ or } x \in A \cap C \Rightarrow x \in (A \cap B) \cup (A \cap C)$.

    $x \in (A \cap B) \cup (A \cap C) \Rightarrow x \in (A \cap B) \text{ or } x \in (A \cap C) \Rightarrow x \in A \text{ and } x \in B \cup C \Rightarrow x \in A \cap (B \cup C)$.
\end{proof}

\begin{lemma}[De mongan laws]~

    !(A or B) = !A and !B.

    !(A and B) = !A or !B.
\end{lemma}

\begin{proof}
    Prop. \ref{prop:sets-boolean-algebra} (\ref{item:sets-boolean-algebra-8}).

    $x \in X \backslash (A \cup B) \Rightarrow x \in X \text{ and } (x \notin A \text{ and } x \notin B)$.

    $x \in X \backslash (A \cap B) \Rightarrow x \in X \text{ and } (x \notin A \text{ or } x \notin B)$.
\end{proof}

\begin{example}~

    $\{x \in A : P(x) \text{ is true}\} \Rightarrow \{f(x) : x \in A; P(x) \text{ is true}\}$.
\end{example}

\begin{example}
    $\{n+3 : n \in \mathbb{N}, 0 \leq n \leq 5\} = \{8-m : m \in \mathbb{N}, 0 \leq m \leq 3\}$.
\end{example}

\begin{example}[Axiom \ref{axiom:sets} (\ref{item:sets-8}) $\Rightarrow$ Axiom \ref{axiom:sets} (\ref{item:sets-7})]~

    $y \in \{x:P(x) \text{ is true for some } x \in A\} \Rightarrow y \in \{x \in A : P(x) \text{ is true}\}$.
\end{example}
